        \subsection{Vector Laplacian}
            \label{VectorLaplacian}
            \begin{definition}
                $A: \bm{r}\in\realNumbers^3 \mapsto A(\bm{r}) \in \realNumbers^3$($A$の各成分は$\mathrm{C}^2$級)の Vector Laplacian $\nabla^2 A$は次式で定義される。
                \[ \nabla^2A \coloneq \nabla(\nabla\cdot A) - \nabla\times(\nabla\times A) \]
            \end{definition}

        \subsection{ベクトル三重積の公式: \texorpdfstring{$A\times(B\times C) = \inprod{A}{C}B - \inprod{A}{B}C$}{A×(B×C) = (A・C)B - (A・B)C}}
            \begin{proof}
                \quad\par
                $A,B,C$を次のようにおく。
                \begin{align*}
                    A &= \bm{i}_1 A_1 + \bm{i}_2 A_2 + \bm{i}_3 A_3 \\
                    B &= \bm{i}_1 B_1 + \bm{i}_2 B_2 + \bm{i}_3 B_3 \\
                    C &= \bm{i}_1 C_1 + \bm{i}_2 C_2 + \bm{i}_3 C_3
                \end{align*}
                すると
                \begin{align*}
                    A\times(B\times C) &= (\bm{i}_1 A_1 + \bm{i}_2 A_2 + \bm{i}_3 A_3) \times \left[\bm{i}_1(B_2 C_3 - B_3 C_2) + \bm{i}_2(B_3 C_1 - B_1 C_3) + \bm{i}_3(B_1 C_2 - B_2 C_1)\right] \\
                    &= \phantom{+}\bm{i}_3 A_1(B_3 C_1 - B_1 C_3) - \bm{i}_2 A_1(B_1 C_2 - B_2 C_1) \\
                    &\phantom{=} -\bm{i}_3 A_2(B_2 C_3 - B_3 C_2) + \bm{i}_1 A_2(B_1 C_2 - B_2 C_1) \\
                    &\phantom{=} +\bm{i}_2 A_3(B_2 C_3 - B_3 C_2) - \bm{i}_1 A_3(B_3 C_1 - B_1 C_3) \\
                    &= \phantom{+}\bm{i}_1 \left[(A_2 C_2 + A_3 C_3)B_1 - (A_2 B_2 + A_3 B_3)C_1\right] \\
                    &\phantom{=} + \bm{i}_2 \left[(A_1 C_1 + A_3 C_3)B_2 - (A_1 B_1 + A_3 B_3)C_2\right] \\
                    &\phantom{=} + \bm{i}_3 \left[(A_1 C_1 + A_2 C_2)B_2 - (A_1 B_1 + A_2 B_2)C_3\right] \\
                    &= \phantom{+}\bm{i}_1 \left[(\inprod{A}{C} - A_1 C_1)B_1 - (\inprod{A}{B} - A_1 B_1)C_1\right] \\
                    &\phantom{=} + \bm{i}_2 \left[(\inprod{A}{C} - A_2 C_2)B_2 - (\inprod{A}{B} - A_2 B_2)C_2\right] \\
                    &\phantom{=} + \bm{i}_3 \left[(\inprod{A}{C} - A_3 C_3)B_3 - (\inprod{A}{B} - A_3 B_3)C_3\right] \\
                    &= \phantom{+}\bm{i}_1 \left[\inprod{A}{C}B_1 - \inprod{A}{B}C_1\right] \\
                    &\phantom{=} + \bm{i}_2 \left[\inprod{A}{C}B_2 - \inprod{A}{B}C_2\right] \\
                    &\phantom{=} + \bm{i}_3 \left[\inprod{A}{C}B_3 - \inprod{A}{B}C_3\right] \\
                    &= \inprod{A}{C}B - \inprod{A}{B}C
                \end{align*}
            \end{proof}

        \subsection{発散の別表現: \texorpdfstring{$\nabla\cdot A = \frac{1}{|V|}\lim_{|V|\to 0} \int_{\partial V} A(\bm{r})\cdot\bm{n}(\bm{r})\mathrm{d}^2\bm{r}$}{発散はベクトル場の法線面積分の体積の商の極限である}}
            \begin{shadebox}
                $\bm{r}_0\in\realNumbers^3$とし，$\Omega\subset\realNumbers^3$上のベクトル場$A$が$\bm{r}_0$の近傍で$C^1$級であるとする。
                $\bm{r}_0$を含む微小な領域を$V$とし，$\partial V$を$V$の境界，$|V|$を$V$の体積とする。
                但し$\partial V$は$C^1$級であるとする。
                このとき次式が成り立つ。
                \[ (\nabla\cdot A)(\bm{r}_0) = \frac{1}{|V|}\lim_{|V|\to 0}\int_{\partial V} A(\bm{r})\cdot\bm{n}(\bm{r})\mathrm{d}^2\bm{r} \]
                ここに$\bm{n}(\bm{r})$は$\bm{r}\in\partial V$に於ける外向きの法線ベクトルである。
            \end{shadebox}
            \begin{proof}
                \quad\par
                発散定理より次式が成り立つ。
                \[ \int_{\partial V} A(\bm{r})\cdot\bm{n}(\bm{r})\mathrm{d}^2\bm{r} = \int_V (\nabla\cdot A)(\bm{r})\mathrm{d}^3\bm{r} \]
                $\varepsilon>0$を任意に小さくとる。
                $A$が$\bm{r}_0$の近傍で$C^1$級であるという仮定から，$V$を充分小さくとれば，$|(\nabla\cdot A)(\bm{r}) - (\nabla\cdot A)(\bm{r}_0)| < \varepsilon$である。
                よって次式が成り立つ。
                \[ \left| \frac{1}{|V|}\int_V (\nabla\cdot A)(\bm{r})\mathrm{d}^3\bm{r} - (\nabla\cdot A)(\bm{r}_0) \right| < \varepsilon \]
                $|V|\to 0$とすれば定理の主張が成り立つ。
            \end{proof}
        \subsection{\texorpdfstring{$\nabla_\bm{r}\cdot f(\bm{r})\bm{A}(\bm{r}) = \bigl(\nabla_\bm{r}f(\bm{r})\bigr)\cdot \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\cdot\bm{A}(\bm{r})$}{ベクトルのスカラー関数倍の発散}}
            \label{ベクトルのスカラー関数倍の発散}
            \begin{shadebox}
                $\bm{r}\in\realNumbers$とする。
                $f: \realNumbers^3\to\complexNumbers,A(\bm{r}): \realNumbers^3\to\realNumbers^3$はともに$\mathrm{C}^1$級とする。
                このとき次式が成り立つ。
                \[ \nabla_\bm{r}\cdot f(\bm{r})\bm{A}(\bm{r}) = \bigl(\nabla_\bm{r}f(\bm{r})\bigr)\cdot \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\cdot\bm{A}(\bm{r}) \]
            \end{shadebox}
            \begin{proof}
                \begin{align*}
                    \nabla_\bm{r}\cdot f(\bm{r})\bm{A}(\bm{r}) &= \sum_{i=1}^3 \partDerivLong{f(\bm{r})\bm{A}_i(\bm{r})}{r_i}{} = \sum_{i=1}^3 \left(\partDeriv{f(\bm{r})}{r_i}{}A_i(\bm{r}) + f(\bm{r})\partDeriv{A_i(\bm{r})}{r_i}{}\right) \\
                    &= \bigl(\nabla_\bm{r}f(\bm{r})\bigr)\cdot \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\cdot\bm{A}(\bm{r})
                \end{align*}
            \end{proof}
        \subsection{\texorpdfstring{$\nabla_\bm{r}\times f(\bm{r})\bm{A}(\bm{r}) = \bigl(\nabla_\bm{r}f(\bm{r})\bigr)\times \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\times\bm{A}(\bm{r})$}{ベクトルのスカラー関数倍の回転}}
            \label{ベクトルのスカラー関数倍の回転}
            \begin{shadebox}
                $\bm{r}\in\realNumbers$とする。
                $f: \realNumbers^3\to\complexNumbers,A(\bm{r}): \realNumbers^3\to\realNumbers^3$はともに$\mathrm{C}^1$級とする。
                このとき次式が成り立つ。
                \[ \nabla_\bm{r}\times f(\bm{r})\bm{A}(\bm{r}) = \bigl(\nabla_\bm{r}f(\bm{r})\bigr)\times \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\times\bm{A}(\bm{r}) \]
            \end{shadebox}
            \begin{proof}
                \quad\par
                $i+j$を3で割った余りを$[i+j]$と表すことにする。
                \begin{align*}
                    \phantom{=} &\nabla_\bm{r}\times f(\bm{r})\bm{A}(\bm{r}) = \sum_{i=1}^3 \bm{i}_i\left(\partDeriv{f(\bm{r})A_{[i+2]}(\bm{r})}{r_{[i+1]}}{} - \partDeriv{f(\bm{r})A_{[i+1]}(\bm{r})}{r_{[i+2]}}{}\right) \\
                    = &\sum_{i=1}^3 \bm{i}_i \left[\left(\partDeriv{f(\bm{r})}{r_{[i+1]}}{}A_{[i+2]}(\bm{r}) - \partDeriv{f(\bm{r})}{r_{[i+2]}}{}A_{[i+3]}(\bm{r})\right) + f(\bm{r})\left(\partDeriv{A_{[i+2]}(\bm{r})}{r_{[i+1]}}{} - \partDeriv{A_{[i+1]}(\bm{r})}{r_{[i+2]}}{}\right)\right] \\
                    = &\bigl(\nabla_\bm{r}f(\bm{r})\bigr)\times \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\times\bm{A}(\bm{r})
                \end{align*}
            \end{proof}
        \subsection{\texorpdfstring{$\nabla_\bm{r}\times\bigl(\bm{C}\times\bm{A}(\bm{r})\bigr) = (\nabla\cdot\bm{A}(\bm{r}))\bm{C} - J_\bm{A}\bm{C}$}{定数ベクトルとの外積の回転}}
            \label{定数ベクトルとの外積の回転}
            \begin{shadebox}
                $\bm{r},\bm{C}\in\realNumbers$とする。
                $A: \realNumbers^3\to\realNumbers^3$は$\mathrm{C}^1$級とする。
                このとき次式が成り立つ。
                \[ \nabla_\bm{r}\times\bigl(\bm{C}\times\bm{A}(\bm{r})\bigr) = (\nabla\cdot\bm{A}(\bm{r}))\bm{C} - J_\bm{A}\bm{C} \]
                ここに$J_\bm{A}$は$A$のJacobi行列である。
            \end{shadebox}
            \begin{proof}
                \begin{align*}
                    &\phantom{=} \nabla_\bm{r}\times\bigl(\bm{C}\times\bm{A}(\bm{r})\bigr) \\
                    &= \bm{i}_1\left[C_1\partDerivLong{A_2}{r_2}{} - C_2\partDerivLong{A_1}{r_2}{} - C_3\partDerivLong{A_1}{r_3}{} + C_1\partDerivLong{A_3}{r_3}{}\right] \\
                    &\phantom{=} + \bm{i}_2\left[C_2\partDerivLong{A_3}{r_3}{} - C_3\partDerivLong{A_2}{r_3}{} - C_1\partDerivLong{A_2}{r_1}{} + C_2\partDerivLong{A_1}{r_1}{}\right] \\
                    &\phantom{=} + \bm{i}_3\left[C_3\partDerivLong{A_1}{r_1}{} - C_1\partDerivLong{A_3}{r_1}{} - C_2\partDerivLong{A_3}{r_2}{} + C_3\partDerivLong{A_2}{r_2}{}\right]
                \end{align*}
                $\bm{i}_1$の係数を変形して次式を得る。
                \[ C_1\left(\partDerivLong{A_1}{r_1}{} + \partDerivLong{A_2}{r_2}{} + \partDerivLong{A_3}{r_3}{}\right) - C_1\partDerivLong{A_1}{r_1}{} - C_2\partDerivLong{A_1}{r_2}{} - C_3\partDerivLong{A_1}{r_3}{} = C_1\nabla\cdot\bm{A} - \bm{C}\cdot\nabla A_1 \]
                $\bm{i}_2, \bm{i}_3$についても同様にして，結局次式を得る。
                \[ \nabla_\bm{r}\times\bigl(\bm{C}\times\bm{A}(\bm{r})\bigr) = (\nabla\cdot\bm{A})\bm{C} - (\bm{C}\cdot\nabla A_1)\bm{i}_1 - (\bm{C}\cdot\nabla A_2)\bm{i}_2 - (\bm{C}\cdot\nabla A_3)\bm{i}_3 = (\nabla\cdot\bm{A}(\bm{r}))\bm{C} - J_\bm{A}\bm{C} \]
            \end{proof}
        \subsection{3次元Euclid空間に於ける Vector Laplacian: \texorpdfstring{$\nabla^2 A = \bm{i}_1 \Delta A_1 + \bm{i}_2 \Delta A_2 + \bm{i}_3 \Delta A_3$}{∇\string^2 A = i1ΔA1 + i2ΔA2 + i3ΔA3}}
            \begin{shadebox}
                \ref{VectorLaplacian}に於いて，特に3次元Euclid空間を考える場合は次式のように簡約化される。
                \[ \nabla^2 A = \bm{i}_1 \Delta A_1 + \bm{i}_2 \Delta A_2 + \bm{i}_3 \Delta A_3 \]
            \end{shadebox}
            \begin{proof}
                \begin{align*}
                    \nabla(\nabla\cdot A) &= \bm{i}_1 \partDerivLong{(\nabla\cdot A)}{r_1}{} + \bm{i}_2 \partDerivLong{(\nabla\cdot A)}{r_2}{} + \bm{i}_3 \partDerivLong{(\nabla\cdot A)}{r_3}{} \\
                    &= \phantom{+} \bm{i}_1\left(\partDerivLong{A_1}{r_1}{2} + \partDerivIIHeteroLong{A_2}{r_1}{r_2} + \partDerivIIHeteroLong{A_3}{r_1}{r_3}\right) \\
                    &\phantom{=} + \bm{i}_2\left(\partDerivIIHeteroLong{A_1}{r_2}{r_1} + \partDerivLong{A_2}{r_2}{2} + \partDerivIIHeteroLong{A_3}{r_2}{r_3}\right) \\
                    &\phantom{=} + \bm{i}_3\left(\partDerivIIHeteroLong{A_1}{r_3}{r_1} + \partDerivIIHeteroLong{A_2}{r_3}{r_2} + \partDerivLong{A_3}{r_3}{2}\right)
                \end{align*}
                \begin{align*}
                    \nabla\times(\nabla\times A) &= \nabla\times\left[\bm{i}_1\left(\partDeriv{A_3}{r_2}{} - \partDeriv{A_2}{r_3}{}\right) + \bm{i}_2\left(\partDeriv{A_1}{r_3}{} - \partDeriv{A_3}{r_1}{}\right) + \bm{i}_3\left(\partDeriv{A_2}{r_1}{} - \partDeriv{A_1}{r_2}{}\right)\right] \\
                    &=\phantom{+} \bm{i}_1\left(-\partDerivLong{A_1}{r_2}{2} - \partDerivLong{A_1}{r_3}{2} + \partDerivIIHeteroLong{A_2}{r_2}{r_1} + \partDerivIIHeteroLong{A_3}{r_3}{r_2}\right) \\
                    &\phantom{=} + \bm{i}_2\left(-\partDerivLong{A_2}{r_3}{2} - \partDerivLong{A_2}{r_1}{2} + \partDerivIIHeteroLong{A_3}{r_3}{r_2} + \partDerivIIHeteroLong{A_1}{r_1}{r_2}\right) \\
                    &\phantom{=} + \bm{i}_3\left(-\partDerivLong{A_3}{r_1}{2} - \partDerivLong{A_3}{r_2}{2} + \partDerivIIHeteroLong{A_1}{r_1}{r_3} + \partDerivIIHeteroLong{A_2}{r_2}{r_3}\right)
                \end{align*}
                $A_1,A_2,A_3$が$\mathrm{C}^2$級であるから偏微分の順序が交換可能であるので，
                \[ \nabla(\nabla\cdot A) - \nabla\times(\nabla\times A) = \bm{i}_1 \Delta A_1 + \bm{i}_2 \Delta A_2 + \bm{i}_3 \Delta A_3 \]
            \end{proof}
        \subsection{\texorpdfstring{$\nabla\cdot(\nabla^2\bm{A}) = \Delta(\nabla\cdot\bm{A})$}{ベクトルLaplacianの発散}}
            \label{ベクトルLaplacianの発散}
            \begin{shadebox}
                $\bm{r}\in\realNumbers$とする。
                $A: \realNumbers^3\to\realNumbers^3$は$\mathrm{C}^3$級とする。
                次式が成り立つ。
                \[ \nabla\cdot(\nabla^2\bm{A}) = \Delta(\nabla\cdot\bm{A}) \]
            \end{shadebox}
            \begin{proof}
                \begin{align*}
                    \nabla\cdot\nabla^2\bm{A} &= \nabla\cdot\left(\bm{i}_1 \Delta A_1 + \bm{i}_2 \Delta A_2 + \bm{i}_3 \Delta A_3\right) = \partDerivLong{\Delta A_1}{r_1}{} + \partDerivLong{\Delta A_2}{r_2}{} + \partDerivLong{\Delta A_3}{r_3}{} \\
                    &= \Delta\left(\partDerivLong{A_1}{r_1}{} + \partDerivLong{A_2}{r_2}{} + \partDerivLong{A_3}{r_3}{}\right) = \Delta(\nabla\cdot\bm{A})
                \end{align*}
            \end{proof}
        \subsection{\texorpdfstring{$\nabla\left(\inprod{\nabla f}{\bm{v}}\right) = \left(\nabla^2f\right)\bm{v}$}{∇(∇f v) = (∇\string^2 f)v}}
            \begin{shadebox}
                $\bm{v}$を定数ベクトルとするとき
                \[\nabla\left(\inprod{\nabla f}{\bm{v}}\right) = \left(\nabla^2f\right)\bm{v}\]
            \end{shadebox}
            \begin{proof}
                \quad\par
                \[\partDerivLong{\inprod{\nabla f}{\bm{v}}}{x_k}{} = \partDerivLong{\sum_{i=1}^{n}{\partDeriv{f}{x_i}{}v_i}}{x_k}{} = \sum_{i=1}^{n}{\partDerivIIHetero{f}{x_k}{x_i}}v_i\]
                であるから
                \[
                    \nabla\left(\inprod{\nabla f}{\bm{v}}\right) = \left[
                        \begin{array}{c}
                            \partDerivLong{\inprod{\nabla f}{\bm{v}}}{x_1}{}\\
                            \partDerivLong{\inprod{\nabla f}{\bm{v}}}{x_2}{}\\
                            \vdots\\
                            \partDerivLong{\inprod{\nabla f}{\bm{v}}}{x_n}{}
                        \end{array}
                    \right] = \left[
                        \begin{array}{c}
                            \left[\partDeriv{f}{x_1}{2}, \partDerivIIHetero{f}{x_1}{x_2},\dotsc,\partDerivIIHetero{f}{x_1}{x_n}\right]\bm{v}\\
                            \vdots\\
                            \left[\partDerivIIHetero{f}{x_n}{x_1}, \partDerivIIHetero{f}{x_n}{x_2},\dotsc,\partDeriv{f}{x_n}{2}\right]\bm{v}
                        \end{array}
                    \right] = \left(\nabla^2f\right)\bm{v}
                \]
            \end{proof}

            \subsubsection{補題: 全空間に渡る積分が存在するための十分条件}
                \label{全空間に渡る積分が存在するための十分条件}
                \begin{shadebox}
                    $\bm{r}\in\realNumbers^3$とする。
                    $f:\realNumbers^3\to\complexNumbers$は十分遠方に於いて，その絶対値がある$\alpha>0$に対して$o(\norm{\bm{r}}_2^{-3-\alpha})$であるとする。
                    このとき$\realNumbers^3$上での$f$の積分は存在する。
                    すなわち次式が成り立つ。
                    \[ R>0, V(R) \coloneq \setComprehension{\bm{r}\in\realNumbers^3}{\norm{\bm{r}}_2 \leq R} ,\lim_{R\to\infty} \abs{\int_{V(R)} f(\bm{r}) \mathrm{d}^3\bm{r}} < \infty \]
                \end{shadebox}
                \begin{proof}
                    \quad\par
                    $R>0$は十分大きいものとし，$I(R)$を次式で定義する。
                    \[ I(R) \coloneq \int_{V(R)} f(\bm{r}) \mathrm{d}^3\bm{r} \]

            \subsubsection{Helmholtzの定理: 任意の\texorpdfstring{$\mathrm{C}^1$}{C\string^1}級のベクトル場は回転が\texorpdfstring{$\bm{0}$}{0}である\texorpdfstring{$\mathrm{C}^1$}{C\string^1}級の場と発散が0である\texorpdfstring{$\mathrm{C}^1$}{C\string^1}級の場に分解できる}
            \begin{proof}
                \quad\par
                ベクトル場 $\vA:\realNumbers^3\to\complexNumbers^3$ は $\mathrm{C}^1$ 級であるとする。
                $\vA$ のFourier変換を $\vB:\realNumbers^3\to\complexNumbers^3$ とする（次式）。
                \begin{gather*}
                    \vB(\vk) = \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vr\in\realNumbers^3}{}{\vA(\vr)\exp\parens*{-i\vk\cdot\vr}}{3}{\vr} \\
                    \vA(\vr) = \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{\vB(\vomega)\exp\parens*{i\vk\cdot\vr}}{3}{\vomega}
                \end{gather*}
                これを用いて $\vA$ の発散と回転を計算すると次式を得る。
                \begin{align*}
                    \parens*{\nabla\cdot\vA}(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{\nabla\cdot\parens*{\vB(\vomega)\exp\parens*{i\vk\cdot\vr}}}{3}{\vomega} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{\parens*{\nabla\exp\parens*{i\vk\cdot\vr}}\cdot\vB(\vomega)}{3}{\vomega} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{i\vomega\cdot\vB(\vomega)\exp\parens*{i\vk\cdot\vr}}{3}{\vomega} \tag{1} \\
                    \parens*{\nabla\times\vA}(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{\nabla\times\parens*{\vB(\vomega)\exp\parens*{i\vk\cdot\vr}}}{3}{\vomega} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{\parens*{\nabla\exp\parens*{i\vk\cdot\vr}}\times\vB(\vomega)}{3}{\vomega} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{i\vomega\times\vB(\vomega)\exp\parens*{i\vk\cdot\vr}}{3}{\vomega} \tag{2}
                \end{align*}
                上式中の $\vB(\omega)$ を $\vomega$ 方向の成分 $\vBP$ とそれに垂直な成分 $\vBO \coloneq \vB(\omega) - \vBP$ に分解する(`P': `parallel', `O': `orthogonal')。
                $\vBP$ を具体的に書くと次式である。
                \[ \vBP = \parens*{\frac{\vomega}{\norm{\vomega}_2}\cdot\vB(\omega)}\frac{\vomega}{\norm{\vomega}_2} = \parens*{\frac{\vomega}{\norm{\vomega}_2^2}\cdot\vB(\vomega)}\vomega \]
                これを用いて $\vA$ を分解する。
                \begin{align*}
                    \vA(\vr) &= \underbrace{\frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{\vBP\exp\parens*{i\vk\cdot\vr}}{3}{\vomega}}_{\text{(A)}} + \underbrace{\frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{\vBO\exp\parens*{i\vk\cdot\vr}}{3}{\vomega}}_{\text{(B)}} \tag{3}\\
                    &= \vARL(\vr) + \vADL(\vr) \quad \text{where} \quad \vARL(\vr) \coloneq \text{(A)}, \quad \vADL(\vr) \coloneq \text{(B)}
                \end{align*}
                `RL', `DL' はそれぞれ `rotation-less', `divergence-less' の意味であるが，その理由は直後に明らかになる。
                \par
                式(1),(2)の計算と全く同様にして次式を得る。
                \begin{align*}
                    \nabla\cdot\vADL(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{i\vomega\cdot\vBO\exp\parens*{i\vk\cdot\vr}}{3}{\vomega} = 0 \\
                    \nabla\times\vARL(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vomega\in\realNumbers^3}{}{i\vomega\times\vBP\exp\parens*{i\vk\cdot\vr}}{3}{\vomega} = \bm{0}
                \end{align*}
                これと式(3)より $(\nabla\cdot\vARL)(\vomega) = \parens*{\nabla\cdot\vA}(\vr),\quad (\nabla\times\vADL)(\vomega) = \parens*{\nabla\times\vA}(\vr)$ も成り立つ。
            \end{proof}

        \subsection{球殻定理}
            \begin{shadebox}
                $R>0,V \coloneq \setComprehension{\bm{r}\in\realNumbers^3}{\norm{\bm{r}}_2\leq R}$とする。
                ベクトル場$\bm{A}(\bm{r})$を次式で定義する。
                \[ \bm{A}(\bm{r}) \coloneq \int_V \frac{\bm{r}-\bm{r}'}{\norm{\bm{r}-\bm{r}'}_2^3} \mathrm{d}^3\bm{r}' \]
                このとき次が成り立つ。
                \[
                    \bm{A}(\bm{r}) = \frac{4}{3}\pi
                    \begin{cases}
                        \bm{r} & (\norm{\bm{r}}_2\leq R) \\
                        \frac{R^3}{\norm{\bm{r}}_2^3}\bm{r} & (\norm{\bm{r}}_2 > R)
                    \end{cases}
                \]
            \end{shadebox}
            \begin{proof}
                \quad\par
                必要なら系全体を回転させ，一般性を失わず$\bm{r} = [0,0,r]^\top\;(r>0)$とする。
                球座標変換を用いて積分を実行する。
                $\bm{r}' \coloneq r'[\sin\theta\cos\phi,\sin\theta\sin\phi,\cos\theta]^\top$とすると
                \begin{align*}
                    \bm{A}(\bm{r}) &= \integrate{0}{R}{\integrate{0}{\pi}{\integrate{0}{2\pi}{\frac{\bm{r}-\bm{r}'}{\norm{\bm{r}-\bm{r}'}_2^3}r'^2(\sin\theta)}{}{\phi}}{}{\theta}}{}{r'} \\
                    &= 2\pi\bm{i}_3\integrate{0}{R}{\integrate{0}{\pi}{\frac{r-r'\cos\theta}{(r^2+r'^2-2rr'\cos\theta)^{3/2}}r'^2(\sin\theta)}{}{\theta}}{}{r'}
                \end{align*}
                ここに$\bm{i}_3$は第3軸方向の単位ベクトルである。
                \par
                $f(r) \coloneq -(r^2+r'^2-2rr'\cos\theta)^{-1/2}r'^2\sin\theta$とすると次式が成り立つ。
                \[ \deriv{f}{r}{}(r) = \frac{(r-r'\cos\theta)r'^2\sin\theta}{(r^2+r'^2-2rr'\cos\theta)^{3/2}} \]
                上式右辺の$(r',\theta)=(0,0)$近傍での振る舞いを考える。
                $r'=r+\Delta r\;(\Delta r\neq 0)$とすると，$\sin\theta = O(\theta),\cos\theta = 1+O(\theta^2)$なので分母・分子ともに$r^3 O(\theta^3)$となる。
                よって積分領域に於いて$f$は$r$の$\mathrm{C}^1$級関数であるから積分と微分の順序交換が可能であり，次式が成り立つ。
                \begin{align*}
                    \bm{A}(\bm{r}) &= 2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\integrate{0}{\pi}{f(r)}{}{\theta}}{}{r'}}{r}{} \\
                    &= -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\integrate{0}{\pi}{\frac{r'^2\sin\theta}{\sqrt{r^2+r'^2-2rr'\cos\theta}}}{}{\theta}}{}{r'}}{r}{}
                \end{align*}
                $g(\theta) \coloneq \frac{r'}{r}\sqrt{r^2+r'^2-2rr'\cos\theta}$とすると次式が成り立つ。
                \[ \deriv{g}{\theta}{}(\theta) = \frac{r'^2\sin\theta}{\sqrt{r^2+r'^2-2rr'\cos\theta}} \]
                よって次式が成り立つ。
                \[ \bm{A}(\bm{r}) = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\left[g(\theta)\right]_0^\pi}{}{r'}}{r}{} = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\frac{r'}{r}\bigl(|r+r'|-|r-r'|\bigr)}{}{r'}}{r}{} \]
                $R<r$のときは次式が成り立つ。
                \[ \bm{A}(\bm{r}) = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\frac{2r'^2}{r}}{}{r'}}{r}{} = \frac{4\pi R^3}{3r^2}\bm{i}_3 = \frac{4\pi R^3}{3}\frac{\bm{r}}{\norm{\bm{r}}_2^3} \]
                $R\geq r$のときは$r'$に関する積分領域を$[0,r]$と$[r,R]$に分けて考える。
                前者については$R<r$のときの結果で$R$を$r$で置き換えたものとなる。
                後者については$|r+r'|-|r-r'| = 2r$となり，次式が成り立つ。
                \[ \bm{A}(\bm{r}) = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{2r'}{}{r'}}{r}{} = \bm{0} \]
            \end{proof}

        \subsection{非負領域の共通部分の非負領域}
            \begin{shadebox}
                $\realNumbers^n$上で考える。$\bm{v}_1,\dotsc,\bm{v}_m\;(m\leq n)$は一次独立であるとし，$\bm{v}$の非負領域を
                \[P(\bm{v})\coloneq\left\{\bm{w}|\inprod{\bm{v}}{\bm{w}}\geq0\right\})\]
                で定義する。あるベクトル$\bm{u}$が$\bigcap_{i=1}^{m}{P(\bm{v}_i)}$の非負領域にある，すなわち
                \[\inprod{\bm{u}}{\bm{w}}\geq0,\forall\bm{w}\in\bigcap_{i=1}^{m}{P(\bm{v}_i)}\]
                であるとき，$\bm{u}$は適当な定数$a_i\geq0(i=1,\dotsc,m)$によって
                \[\bm{u} = \sum_{i=1}^{m}{a_i\bm{v}_i}\]
                と表せる。
            \end{shadebox}
            \begin{proof}
                \quad\par
                    まず$\bm{u}$が$\bm{v}_1,\dotsc,\bm{v}_m$の一次結合で表せることを示す。\\
                    $W \coloneq \Span{\bm{v}_1,\dotsc,\bm{v}_m}$とすると$\realNumbers^n = W\oplus W^\bot$であるから，適当な定数$c_1,\dotsc,c_m$及び適当なベクトル$\bm{w}_t\in W^\bot$を用いて
                    \begin{equation}
                        \bm{u} = \sum_{i=1}^{m}{c_i\bm{v}_i} + \bm{w}_t
                    \end{equation}
                    と表せる。$\bm{p}$を$\bigcap_{i=1}^{m}{P(\bm{v}_i)}$の任意のベクトルとすると
                    \[\inprod{\bm{v}_i}{\bm{p}}\geq0,\forall\bm{v}_i\in W\]
                    また，$\bm{w}_{t2}$を$W^\bot$の任意のベクトルとすると
                    \[\inprod{\bm{v}_i}{\bm{w}_{t2}}=0,\forall\bm{v}_i\in W\]
                    従って$\bm{p}' \coloneq \bm{p}+\bm{w}_{t2}$とすると
                    \[\inprod{\bm{v}_i}{\bm{p}'}\geq0,\forall\bm{v}_i\in W\]
                    であるから$\bm{p'}$もまた$\bigcap_{i=1}^{m}{P(\bm{v}_i)}$のベクトルである。従って$\inprod{\bm{u}}{\bm{p}'}\geq0$であるから
                    \[0\leq\inprod{\bm{u}}{\bm{p}'} = \inprod{\sum_{i=1}^{m}{c_i\bm{v}_i}}{\bm{p}} + \inprod{\bm{w}_{t1}}{\bm{w}_{t2}}\]
                    である。然るに$\bm{w}_{t2}\in W^\bot$は任意であるから，$\bm{w}_{t1}=\bm{0}$でない限り$\inprod{\bm{w}_{t1}}{\bm{w}_{t2}}$の値をいくらでも小さくできる。
                    よって上式が成り立つためには$\bm{w}_{t1}=\bm{0}$でなくてはならならず，式(1)より結局
                    \[\bm{u} = \sum_{i=1}^{m}{c_i\bm{v}_i}\]
                    と表せることになる。
                    \par
                    次に$c_i$が全て非負であることを示す。%そのために$\left(\bm{d}_i\in\bigcap_{i=1}^{m}{P(\bm{v}_i)}\right)\land\inprod{\bm{v}_j}{\bm{d}_i}=\delta_{ij}\;(\delta_{ij}\text{はクロネッカーのデルタ})$なるベクトル$\bm{d}_i$が存在することを示そう。
                    $\bm{v}_1,\dotsc,\bm{v}_m\;(m\leq n)$は一次独立であったから次の行列
                    \[
                        A = \left[
                            \begin{array}{c}
                                {\bm{v}_1}^\textrm{T}\\
                                {\bm{v}_2}^\textrm{T}\\
                                \vdots\\
                                {\bm{v}_m}^\textrm{T}
                            \end{array}
                        \right]
                    \]
                    の階数は$m$である。従って次の方程式
                    \[
                        A\bm{x} = \left[
                            \begin{array}{c}
                                0\\
                                \vdots\\
                                1\\
                                \vdots\\
                                0
                            \end{array}
                        \right]
                        \begin{array}{c}
                            \quad\\
                            \quad\\
                            \leftarrow i\\
                            \quad\\
                            \quad
                        \end{array}
                    \]
                    は解を持つ。この解を$\bm{d}_i$とすれば$\inprod{\bm{v}_j}{\bm{d}_i}=\delta_{ij}$であるから$\bm{d}_i\in\bigcap_{i=1}^{m}{P(\bm{v}_i)}$。
                    よって
                    \[0\leq\inprod{\bm{u}}{\bm{d}_i} = c_i\]
            \end{proof}
    \subsection{Stokesの定理}
        \begin{shadebox}
            $c$を空間上の任意の$C^1$級の閉曲線，$S$を，$c$を周$\partial S$とする任意の$C^2$級の曲面, $A$を$C^1$級のベクトル場とするとき
            \[
                \oint_c{A\cdot\dvc} = \iint_S{\nabla\times A}\cdot\dS
            \]
        \end{shadebox}
        \begin{proof}
            \quad\par
            $S$が$S(x,y)$の如く$x,y$の関数で表せる場合について証明すれば十分である。
            なぜなら，$S$が複雑な場合は$S(s,y),S(y,z),S(z,x)$いずれかの型の2変数関数で表せる曲面に分割して，各部分で，これから証明する$S(x,y)$型の場合と同じ手続きを踏んで最後に合計すれば全体の結果になるからである。
            領域分割を行う場合は分割境界でも線積分を行うことになるが，その回数は2で，互いに逆向きであるから，合計する段階で相殺して0になるため最終的な結果には現れない。

                \centering
                \includegraphics[keepaspectratio, scale=0.4]
                {\currfiledir/figs/Stokes-theorem/divide.pdf}
                \caption{$S$の分割}
            \end{figure}
            以降$S$は$S(x,y)$型の曲面とする。$\partial S$を$t$をパラメータとする閉曲線とする。
            すなわち
            \[\partial S\;:\;x=\phi(t), y=\psi(t), t\;:\;0\rightarrow1\]
            また，$xy$平面への$S$の射影を$\Omega$とする。
            \begin{figure}[H]
                \centering
                \includegraphics[keepaspectratio, scale=0.4]
                {\currfiledir/figs/Stokes-theorem/illust.pdf}
                \caption{状況}
            \end{figure}
            では証明しよう。
            線積分で表された定理の左辺を変形して面積分に置き換えて，定理の右辺と見比べる流れで行く。
            \par
            関数の引数を全部書いていてはスペースがいくらあっても足りないので省略して書くことになる(例えば$A_x(x,y,z)$を単に$A_x$と書いたり)のだが，こうすると独立変数と従属変数の関係を見誤る危険が生じるため注意が必要だ。
            迂闊に計算すると罠にはまる。
            特に危ない所では敢えて引数を明示する。
            \begin{align*}
                \oint_c{A\cdot\dvc} &= \oint_{\partial S}{\left(\vix A_x+\viy A_y+\viz A_z\right)\cdot\left(\vix \dx+\viy \dy+\viz \dz\right)} \\
                &= \oint_{\partial S}{\left(A_x\dx+A_y\dy\right)} + \oint_{\partial S}{A_z\dz} \\
                &= \underbrace{\oint_{\partial \textcolor{red}{\Omega}}{\left(A_x(x,y,S(x,y))\dx+A_y(x,y,S(x,y))\dy\right)}}_{\ctext{A}} + \underbrace{\oint_{\partial S}{A_z\dz}}_{\ctext{B}}
            \end{align*}
            \ctext{A}で積分経路が$\partial S$から$\partial\Omega$に変更されたことに注意。Greenの定理を使いたいから$xy$平面に落としたのだ。
            Greenの定理より
            \[
                \ctext{A} = \iint_\Omega{\left[\underbrace{\partDerivLong{A_y(x,y,S(x,y))}{x}{}}_{\ctext{C}}-\underbrace{\partDerivLong{A_x(x,y,S(x,y))}{y}{}}_{\ctext{D}}\right]\dx\dy}
            \]
            \ctext{C}が罠だ。引数を明示していないと$\left.\partDeriv{A_y}{x}{}\right|_{(x,y,z)=(x,y,S(x,y))}$と勘違いしてしまう。
            \newline
            \textcolor{red}{$\partDerivLong{A_y(x,y,S(x,y))}{x}{}$と$\left.\partDeriv{A_y}{x}{}\right|_{(x,y,z)=(x,y,S(x,y))}$は別物}であることに注意せなばならぬ。
            連鎖律に注意して
            \begin{align*}
                \ctext{C} &= \left.\partDeriv{A_y}{x}{}\right|_{(x,y,z)=(x,y,S(x,y))}\partDeriv{x}{x}{} + \left.\partDeriv{A_y}{y}{}\right|_{(x,y,z)=(x,y,S(x,y))}\partDeriv{y}{x}{} + \left.\partDeriv{A_y}{z}{}\right|_{(x,y,z)=(x,y,S(x,y))}\partDeriv{S}{x}{} \\
                &= \partDeriv{A_y}{x}{} + \partDeriv{A_y}{z}{}\partDeriv{S}{x}{}
            \end{align*}
            同様にして
            \[\ctext{D} = \partDeriv{A_x}{y}{} + \partDeriv{A_x}{z}{}\partDeriv{S}{y}{}\]
            よって結局
            \begin{align*}
                \ctext{A} &= \iint_\Omega{\left[\partDeriv{A_y}{x}{} + \partDeriv{A_y}{z}{}\partDeriv{S}{x}{} - \partDeriv{A_x}{y}{} - \partDeriv{A_x}{z}{}\partDeriv{S}{y}{}\right]\dx\dy} \\
                &= \iint_\Omega{\left[\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right) + \partDeriv{A_y}{z}{}\partDeriv{S}{x}{} - \partDeriv{A_x}{z}{}\partDeriv{S}{y}{}\right]\dx\dy}
            \end{align*}
            次に\ctext{B}を見る。$t$による偏微分を$\dot{}$で表すことにする。線積分の定義に従って
            \begin{align*}
                \ctext{B} &= \integrate{0}{1}{A_z(\phi(t),\psi(t),S(\phi(t),\psi(t))\left(\partDeriv{S}{x}{}\dot{\phi(t)}+\partDeriv{S}{y}{}\dot{\psi(t)}\right)}{}{t} \\
                &= \integrate{0}{1}{\left(A_z(\phi(t),\psi(t),S(\phi(t),\psi(t))\partDeriv{S}{x}{}\right)\dot{\phi(t)}}{}{t} + \integrate{0}{1}{\left(A_z(\phi(t),\psi(t),S(\phi(t),\psi(t))\partDeriv{S}{y}{}\right)\dot{\psi(t)}}{}{t} \\
                &= \oint_{\partial \Omega}{A_z(x,y,S(x,y))\partDeriv{S}{x}{}\dx} + \oint_{\partial \Omega}{A_z(x,y,S(x,y))\partDeriv{S}{y}{}\dy} \\
                &= \oint_{\partial \Omega}{\left(A_z(x,y,S(x,y))\partDeriv{S}{x}{}\dx + A_z(x,y,S(x,y))\partDeriv{S}{y}{}\dy\right)}
            \end{align*}
            これにGreenの定理を適用して，連鎖律に注意して偏微分していく。
            \begin{align*}
                \ctext{B} &= \iint_\Omega{\left[\partDerivLong{\left(A_z(x,y,S(x,y))\partDeriv{S}{y}{}\right)}{x}{} - \partDerivLong{\left(A_z(x,y,S(x,y))\partDeriv{S}{x}{}\right)}{y}{}\right]\dx\dy} \\
                &= \iint_\Omega{\left(\partDeriv{A_z}{x}{}\partDeriv{S}{y}{} + \partDeriv{A_z}{z}{}\partDeriv{S}{x}{}\partDeriv{S}{y}{} + A_z\partDerivIIHetero{S}{y}{x} - \partDeriv{A_z}{y}{}\partDeriv{S}{x}{} - \partDeriv{nA_z}{z}{}\partDeriv{S}{y}{}\partDeriv{S}{x}{} - A_z\partDerivIIHetero{S}{x}{y}\right)\dx\dy} \\
                &= \iint_\Omega{\left(\partDeriv{A_z}{x}{}\partDeriv{S}{y}{} - \partDeriv{A_z}{y}{}\partDeriv{S}{x}{}\right)\dx\dy}\quad\left(\because\;\text{$S$は$C^2$級と仮定したから$\partDerivIIHetero{S}{y}{x} = \partDerivIIHetero{S}{x}{y}$}\right)
            \end{align*}
            以上より
            \begin{align*}
                \oint_c{A\cdot\dvc} &= \iint_\Omega{\left[-\left(\partDeriv{A_z}{y}{}-\partDeriv{A_y}{z}{}\right)\partDeriv{S}{x}{}-\left(\partDeriv{A_x}{z}{}-\partDeriv{A_z}{x}{}\right)\partDeriv{S}{y}{}+\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right)\right]\dx\dy}
            \end{align*}
            \par
            これが定理の右辺と等しいことを示せば良い。
            まず$\dS$の正体を暴こう。感覚的には，$\dS$とは，$S$上で$x,y$が微小量$\dx,\dy$だけ動いた時に舐める$S$の面積をノルムに持ち，向きはその微小面上の任意の代表点での上向きの法線ベクトルに一致するようなベクトルであった。
            所謂「面積ベクトル」である。
            \par
            厳密にはRiemann和を取る段階で定義され，$S$上の点$(x,y,S(x,y))$における接平面のうち，その射影が$xy$平面上の矩形$\dx\dy$に収まる部分(3次元上の平行四辺形)の面積をノルムに持ち，向きは接平面の上向き法線ベクトルと一致するようなベクトルである。
            \[\dS = \left(\vix\dx+\viz\partDeriv{S}{x}{}\dx\right)\times\left(\viy\dy+\viz\partDeriv{S}{y}{}\dy\right) = \left(-\vix\partDeriv{S}{x}{} - \viy\partDeriv{S}{y}{} + \viz\right)\dx\dy\]
            よって定理の右辺は
            \begin{align*}
                \iint_S{\nabla\times A}\cdot\dS &= \iint_\Omega{\left[\vix\left(\partDeriv{A_z}{y}{}-\partDeriv{A_y}{z}{}\right)+\viy\left(\partDeriv{A_x}{z}{}-\partDeriv{A_z}{x}{}\right)+\viz\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right)\right]\cdot\dS} \\
                &= \iint_\Omega{\left[-\left(\partDeriv{A_z}{y}{}-\partDeriv{A_y}{z}{}\right)\partDeriv{S}{x}{}-\left(\partDeriv{A_x}{z}{}-\partDeriv{A_z}{x}{}\right)\partDeriv{S}{y}{}+\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right)\right]\dx\dy} \\
                &= \oint_c{A\cdot\dvc}
            \end{align*}
        \end{proof}
