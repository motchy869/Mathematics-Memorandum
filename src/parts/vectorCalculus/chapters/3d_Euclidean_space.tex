        \subsection{計量係数と単位ベクトル}
            \label{計量係数と単位ベクトル}
            $T:\realNumbers^3\to\EuclideanSpace^3$ は可逆であり $\mathrm{C}^1$ 級であるとする。
            $T$ が定める座標系を $S_T$ とする。
            $T$ のJacobi行列の列は互いに直交するものとする。
            $\bm{u}\in\realNumbers^3,\bm{r} = T(\bm{u})$ とする。
            例えば球座標系では $\bm{u} = (r,\theta,\varphi)$ であり，$\bm{r} = \bm{i}_1 r_1 + \bm{i}_2 r_2 + \bm{i}_3 r_3,(r_1,r_2,r_3) = (r\sin\theta\cos\varphi,r\sin\theta\sin\varphi,r\cos\theta)$ である。
            次式で定める量を座標系 $S_T$ の計量係数(Lam\'{e} coefficient)と呼ぶ。
            \[ h_i \coloneq \norm{\partDeriv{\bm{r}}{u_i}{}}_2 \quad (i=1,2,3) \]
            $h_i$ は $T$ の Jacobi 行列 $J$ の第 $i$ 列のノルムであり，逆写像定理から $J$ は可逆なので $h_i \neq 0$ である。
            \par
            次式で定める単位ベクトルを $S_T$ の $u_i$ 方向単位ベクトルと呼ぶ。
            \[ \bm{j}_i := \frac{1}{h_i}\partDeriv{\bm{r}}{u_i}{}\in\EuclideanSpace^3 \]
            $T$ のJacobi行列の列が互いに直交するため，$\bm{j}_1,\bm{j}_2,\bm{j}_3$ は互いに直交する。
        \subsection{単位ベクトルの変換}
            \label{単位ベクトルの変換}
            \begin{shadebox}
                既出の記号の定義を引き継ぐ。
                3 次元 Euclid 空間の標準単位ベクトル $\bm{i}_1,\bm{i}_2,\bm{i}_3$ は直交座標系の単位ベクトル $\bm{j}_1,\bm{j}_2,\bm{j}_3$ を用いて次のように表せる。
                \[ \bm{i}_i = \sum_{j=1}^3 \frac{1}{h_j}\frac{\partial r_i}{\partial u_j}\bm{j}_j \quad (i=1,2,3) \]
                つまり行列を用いて次のように表せる。
                \[ [\bm{i}_1, \bm{i}_2, \bm{i}_3] = [\bm{j}_1, \bm{j}_2, \bm{j}_3]JH^{-1} \]
                ここに $J$ は $T$ のJacobi行列であり，$H := \diag{h_1,h_2,h_3}$ である。
            \end{shadebox}
            \begin{proof}
                \quad\par
                $\bm{j}_1,\bm{j}_2,\bm{j}_3$ は正規直交基底であるから次式が成り立つ。
                \[ \bm{i}_i = \sum_{j=1}^3 (\bm{i}_i\cdot\bm{j}_j)\bm{j}_j = \sum_{j=1}^3 \left(\bm{i}_i\cdot\frac{1}{h_j}\frac{\partial \bm{r}}{\partial u_j}\right)\bm{j}_j = \sum_{j=1}^3 \left(\bm{i}_i\cdot\frac{1}{h_j}\sum_{k=1}^3\frac{\partial r_k}{\partial u_j}\bm{i}_k\right)\bm{j}_j \\
                = \sum_{j=1}^3 \frac{1}{h_j}\frac{\partial r_i}{\partial u_j}\bm{j}_j \]
            \end{proof}
        \subsection{ベクトルの成分の変換}
            \label{ベクトルの成分の変換}
            \begin{shadebox}
                既出の記号の定義を引き継ぐ。
                3次元Euclid空間上のベクトル$\bm{A}(\bm{r}) = \bm{i}_1 A_1(\bm{r}) + \bm{i}_2 A_2(\bm{r}) + \bm{i}_3 A_3(\bm{r})$を$u_1,u_2,u_3$直交座標系で表したものを$\bm{j}_1 \tilde{A}_1(\bm{u}) + \bm{j}_2 \tilde{A}_2(\bm{u}) + \bm{j}_3 \tilde{A}_3(\bm{u})$とするとき，$A_1,A_2,A_3$と$\tilde{A}_1,\tilde{A}_2,\tilde{A}_3$の間には次の関係式が成り立つ。
                \[ [A_1,A_2,A_3]^\top = JH^{-1}[\tilde{A}_1,\tilde{A}_2,\tilde{A}_3]^\top \]
                ここに$J$は$T$のJacobi行列であり，$H := \diag{h_1,h_2,h_3}$である。
            \end{shadebox}
            \begin{proof}
                \begin{align*}
                    \phantom{=} &[\bm{i}_1,\bm{i}_2,\bm{i}_3][A_1,A_2,A_3] = \bm{A}(\bm{r}) = \bm{j}_1 \tilde{A}_1 + \bm{j}_2 \tilde{A}_2 + \bm{j}_3 \tilde{A}_3 = [\bm{j}_1,\bm{j}_2,\bm{j}_3][\tilde{A}_1,\tilde{A}_2,\tilde{A}_3]^\top \\
                    = &[\bm{i}_1,\bm{i}_2,\bm{i}_3]JH^{-1}[\tilde{A}_1,\tilde{A}_2,\tilde{A}_3]^\top
                \end{align*}
            \end{proof}

        \subsection{ベクトル三重積の公式: \texorpdfstring{$A\times(B\times C) = \inprod{A}{C}B - \inprod{A}{B}C$}{A×(B×C) = (A・C)B - (A・B)C}}
            \begin{proof}
                \quad\par
                直交座標系をとり，単位ベクトルを $\bm{j}_1,\bm{j}_2,\bm{j}_3$ とする。
                $A,B,C$を次のようにおく。
                \begin{align*}
                    A &= \bm{j}_1 A_1 + \bm{j}_2 A_2 + \bm{j}_3 A_3 \\
                    B &= \bm{j}_1 B_1 + \bm{j}_2 B_2 + \bm{j}_3 B_3 \\
                    C &= \bm{j}_1 C_1 + \bm{j}_2 C_2 + \bm{j}_3 C_3
                \end{align*}
                すると
                \begin{align*}
                    A\times(B\times C) &= (\bm{j}_1 A_1 + \bm{j}_2 A_2 + \bm{j}_3 A_3) \times \left[\bm{j}_1(B_2 C_3 - B_3 C_2) + \bm{j}_2(B_3 C_1 - B_1 C_3) + \bm{j}_3(B_1 C_2 - B_2 C_1)\right] \\
                    &= \phantom{+}\bm{j}_3 A_1(B_3 C_1 - B_1 C_3) - \bm{j}_2 A_1(B_1 C_2 - B_2 C_1) \\
                    &\phantom{=} -\bm{j}_3 A_2(B_2 C_3 - B_3 C_2) + \bm{j}_1 A_2(B_1 C_2 - B_2 C_1) \\
                    &\phantom{=} +\bm{j}_2 A_3(B_2 C_3 - B_3 C_2) - \bm{j}_1 A_3(B_3 C_1 - B_1 C_3) \\
                    &= \phantom{+}\bm{j}_1 \left[(A_2 C_2 + A_3 C_3)B_1 - (A_2 B_2 + A_3 B_3)C_1\right] \\
                    &\phantom{=} + \bm{j}_2 \left[(A_1 C_1 + A_3 C_3)B_2 - (A_1 B_1 + A_3 B_3)C_2\right] \\
                    &\phantom{=} + \bm{j}_3 \left[(A_1 C_1 + A_2 C_2)B_3 - (A_1 B_1 + A_2 B_2)C_3\right] \\
                    &= \phantom{+}\bm{j}_1 \left[(\inprod{A}{C} - A_1 C_1)B_1 - (\inprod{A}{B} - A_1 B_1)C_1\right] \\
                    &\phantom{=} + \bm{j}_2 \left[(\inprod{A}{C} - A_2 C_2)B_2 - (\inprod{A}{B} - A_2 B_2)C_2\right] \\
                    &\phantom{=} + \bm{j}_3 \left[(\inprod{A}{C} - A_3 C_3)B_3 - (\inprod{A}{B} - A_3 B_3)C_3\right] \\
                    &= \phantom{+}\bm{j}_1 \left[\inprod{A}{C}B_1 - \inprod{A}{B}C_1\right] \\
                    &\phantom{=} + \bm{j}_2 \left[\inprod{A}{C}B_2 - \inprod{A}{B}C_2\right] \\
                    &\phantom{=} + \bm{j}_3 \left[\inprod{A}{C}B_3 - \inprod{A}{B}C_3\right] \\
                    &= \inprod{A}{C}B - \inprod{A}{B}C
                \end{align*}
            \end{proof}

        \subsection{発散の別表現: \texorpdfstring{$\nabla\cdot A = \frac{1}{|V|}\lim_{|V|\to 0} \int_{\partial V} A(\bm{r})\cdot\bm{n}(\bm{r})\mathrm{d}^2\bm{r}$}{発散はベクトル場の法線面積分の体積の商の極限である}}
            \begin{shadebox}
                $\bm{r}_0\in\EuclideanSpace^3$とし，$\Omega\subset\EuclideanSpace^3$上のベクトル場$A$が$\bm{r}_0$の近傍で$C^1$級であるとする。
                $\bm{r}_0$を含む微小な領域を$V$とし，$\partial V$を$V$の境界，$|V|$を$V$の体積とする。
                但し$\partial V$は$C^1$級であるとする。
                このとき次式が成り立つ。
                \[ (\nabla\cdot A)(\bm{r}_0) = \frac{1}{|V|}\lim_{|V|\to 0}\int_{\partial V} A(\bm{r})\cdot\bm{n}(\bm{r})\mathrm{d}^2\bm{r} \]
                ここに$\bm{n}(\bm{r})$は$\bm{r}\in\partial V$に於ける外向きの法線ベクトルである。
            \end{shadebox}
            \begin{proof}
                \quad\par
                発散定理より次式が成り立つ。
                \[ \int_{\partial V} A(\bm{r})\cdot\bm{n}(\bm{r})\mathrm{d}^2\bm{r} = \int_V (\nabla\cdot A)(\bm{r})\mathrm{d}^3\bm{r} \]
                $\varepsilon>0$を任意に小さくとる。
                $A$が$\bm{r}_0$の近傍で$C^1$級であるという仮定から，$V$を充分小さくとれば，$|(\nabla\cdot A)(\bm{r}) - (\nabla\cdot A)(\bm{r}_0)| < \varepsilon$である。
                よって次式が成り立つ。
                \[ \left| \frac{1}{|V|}\int_V (\nabla\cdot A)(\bm{r})\mathrm{d}^3\bm{r} - (\nabla\cdot A)(\bm{r}_0) \right| < \varepsilon \]
                $|V|\to 0$とすれば定理の主張が成り立つ。
            \end{proof}
        \subsection{ベクトルのスカラー関数倍の発散\texorpdfstring{: $\nabla_\bm{r}\cdot f(\bm{r})\bm{A}(\bm{r}) = \bigl(\nabla_\bm{r}f(\bm{r})\bigr)\cdot \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\cdot\bm{A}(\bm{r})$}{}}
            \begin{shadebox}
                $\bm{r}\in\EuclideanSpace^3$とする。
                $f:\EuclideanSpace^3\to\complexNumbers,A:\EuclideanSpace^3\to\EuclideanSpace^3$はともに$\mathrm{C}^1$級とする。
                座標系の取り方に依らず次式が成り立つ。
                \[ \nabla_\bm{r}\cdot f(\bm{r})\bm{A}(\bm{r}) = \parens*{\nabla_\bm{r}f(\bm{r})}\cdot \bm{A}(\bm{r}) + f(\bm{r})\nabla_\bm{r}\cdot\bm{A}(\bm{r}) \]
            \end{shadebox}
            \begin{proof}
                \quad\par
                \ref{空間と座標系の関係}の\ref{Euclid 空間上の，座標が陽に現れない関係式が成り立つ必要十分条件} より，直交座標系上で成立を示せば十分である。
                $\bm{u}\in\realNumbers^3,\bm{r}=T(\bm{u}),\tilde{f}(\bm{u})\coloneq f(\bm{r}),\tilde{A}(\bm{u})\coloneq A(\bm{r})$ とする。
                \begin{align*}
                    \nabla_\bm{r}\cdot f(\bm{r})\bm{A}(\bm{r}) &= \frac{1}{h_1 h_2 h_3}\sum_{i=1}^3 \partDerivLong{\frac{h_1 h_2 h_3}{h_i}\tilde{f} \tilde{A}_i}{u_i}{} = \frac{1}{h_1 h_2 h_3}\sum_{i=1}^3\parens*{\partDeriv{\tilde{f}}{u_i}{}\frac{h_1 h_2 h_3}{h_i} \tilde{A}_i + \tilde{f}\partDerivLong{\frac{h_1 h_2 h_3}{h_i} \tilde{A}_i}{u_i}{}} \\
                    &= \sum_{i=1}^3 \frac{1}{h_i}\partDeriv{\tilde{f}}{u_i}{}\tilde{A}_i + \tilde{f}\frac{1}{h_1 h_2 h_3}\sum_{i=1}^3 \partDerivLong{\frac{h_1 h_2 h_3}{h_i}\tilde{A}_i}{u_i}{} = (\nabla f)\cdot\bm{A} + f\nabla\cdot\bm{A}
                \end{align*}
            \end{proof}

                $\bm{u}\in\realNumbers^3,\bm{r}=T(\bm{u}),\tilde{f}(\bm{u})\coloneq f(\bm{r}),\tilde{A}(\bm{u})\coloneq A(\bm{r})$ とする。

                    \nabla_\bm{r}\times f(\bm{r})\bm{A}(\bm{r}) &= \sum_{i=1}^3 \bm{j}_i\varepsilon_{ijk}\frac{1}{h_j h_k}\partDerivLong{h_k \tilde{f} \tilde{A}_k}{u_j}{} = \sum_{i=1}^3 \bm{j}_i\varepsilon_{ijk}\frac{1}{h_j h_k}\parens*{\partDeriv{\tilde{f}}{u_j}{}h_k \tilde{A}_k + \tilde{f}\partDeriv{h_k \tilde{A}_k}{u_j}{}} \\
                    &= \sum_{i=1}^3 \bm{j}_i\varepsilon_{ijk}\frac{1}{h_j}\partDeriv{\tilde{f}}{u_j}{} \tilde{A}_k + \tilde{f}\sum_{i=1}^3 \bm{j}_i\varepsilon_{ijk}\partDeriv{h_k \tilde{A}_k}{u_j}{} = (\nabla f)\times\bm{A} + f\nabla\times\bm{A}
                \end{align*}
            \end{proof}

            \begin{shadebox}

                座標系の取り方に依らず次式が成り立つ。
                \[ \nabla_\bm{r}\norm{\bm{r}+\bm{a}}^n = n\norm{\bm{r}+\bm{a}}^{n-2}(\bm{r}+\bm{a}) \]
            \end{shadebox}
            \begin{proof}
                \quad\par
                \ref{空間と座標系の関係}の\ref{Euclid 空間上の，座標が陽に現れない関係式が成り立つ必要十分条件} より，直交座標系上で成立を示せば十分である。

                \end{align*}
            \end{proof}

            \subsubsection{Helmholtzの定理: 任意の\texorpdfstring{$\mathrm{C}^1$}{C\string^1}級のベクトル場は回転が\texorpdfstring{$\bm{0}$}{0}である\texorpdfstring{$\mathrm{C}^1$}{C\string^1}級の場と発散が0である\texorpdfstring{$\mathrm{C}^1$}{C\string^1}級の場に分解できる}
            \newcommand*{\vBO}{\bm{B}_\text{O}(\vk)}
            \newcommand*{\vBP}{\bm{B}_\text{P}(\vk)}
            \begin{proof}
                \quad\par
                \ref{空間と座標系の関係}の\ref{Euclid 空間上の，座標が陽に現れない関係式が成り立つ必要十分条件} より，直角座標系上で成立を示せば十分である。
                ベクトル場 $\vA:\realNumbers^3\to\complexNumbers^3$ は $\mathrm{C}^1$ 級であるとする。
                $\vA$ のFourier変換を $\vB:\realNumbers^3\to\complexNumbers^3$ とする（次式）。
                \begin{gather*}
                    \vB(\vk) = \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vr\in\realNumbers^3}{}{\vA(\vr)\exp\parens*{-i\vk\cdot\vr}}{3}{\vr} \\
                    \vA(\vr) = \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{\vB(\vk)\exp\parens*{i\vk\cdot\vr}}{3}{\vk}
                \end{gather*}
                これを用いて $\vA$ の発散と回転を計算すると次式を得る。
                \begin{align*}
                    \parens*{\nabla\cdot\vA}(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{\nabla\cdot\parens*{\vB(\vk)\exp\parens*{i\vk\cdot\vr}}}{3}{\vk} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{\parens*{\nabla\exp\parens*{i\vk\cdot\vr}}\cdot\vB(\vk)}{3}{\vk} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{i\vk\cdot\vB(\vk)\exp\parens*{i\vk\cdot\vr}}{3}{\vk} \tag{1} \\
                    \parens*{\nabla\times\vA}(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{\nabla\times\parens*{\vB(\vk)\exp\parens*{i\vk\cdot\vr}}}{3}{\vk} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{\parens*{\nabla\exp\parens*{i\vk\cdot\vr}}\times\vB(\vk)}{3}{\vk} \\
                    &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{i\vk\times\vB(\vk)\exp\parens*{i\vk\cdot\vr}}{3}{\vk} \tag{2}
                \end{align*}
                上式中の $\vB(\vk)$ を $\vk$ 方向の成分 $\vBP$ とそれに垂直な成分 $\vBO \coloneq \vB(\vk) - \vBP$ に分解する(`P': `parallel', `O': `orthogonal')。
                $\vBP$ を具体的に書くと次式である。
                \[ \vBP = \parens*{\frac{\vk}{\norm{\vk}_2}\cdot\vB(\vk)}\frac{\vk}{\norm{\vk}_2} = \parens*{\frac{\vk}{\norm{\vk}_2^2}\cdot\vB(\vk)}\vk \]
                これを用いて $\vA$ を分解する。
                \begin{align*}
                    \vA(\vr) &= \underbrace{\frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{\vBP\exp\parens*{i\vk\cdot\vr}}{3}{\vk}}_{\text{(A)}} + \underbrace{\frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{\vBO\exp\parens*{i\vk\cdot\vr}}{3}{\vk}}_{\text{(B)}} \tag{3}\\
                    &= \vARL(\vr) + \vADL(\vr) \quad \text{where} \quad \vARL(\vr) \coloneq \text{(A)}, \quad \vADL(\vr) \coloneq \text{(B)}
                \end{align*}
                `RL', `DL' はそれぞれ `rotation-less', `divergence-less' の意味であるが，その理由は直後に明らかになる。
                \par
                式(1),(2)の計算と全く同様にして次式を得る。
                \begin{align*}
                    \nabla\cdot\vADL(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{i\vk\cdot\vBO\exp\parens*{i\vk\cdot\vr}}{3}{\vk} = 0 \\
                    \nabla\times\vARL(\vr) &= \frac{1}{\sqrt{(2\pi)^3}}\integrate{\vk\in\realNumbers^3}{}{i\vk\times\vBP\exp\parens*{i\vk\cdot\vr}}{3}{\vk} = \bm{0}
                \end{align*}
                これと式(3)より $(\nabla\cdot\vARL)(\vk) = \parens*{\nabla\cdot\vA}(\vr),\quad (\nabla\times\vADL)(\vk) = \parens*{\nabla\times\vA}(\vr)$ も成り立つ。
            \end{proof}

            \begin{shadebox}
                $R>0,V \coloneq \setComprehension{\bm{r}\in\EuclideanSpace^3}{\norm{\bm{r}}_2\leq R}$とする。
                ベクトル場$\bm{A}(\bm{r})$を次式で定義する。
                \[ \bm{A}(\bm{r}) \coloneq \int_V \frac{\bm{r}-\bm{r}'}{\norm{\bm{r}-\bm{r}'}_2^3} \mathrm{d}^3\bm{r}' \]
                このとき次が成り立つ。
                \[
                    \bm{A}(\bm{r}) = \frac{4}{3}\pi
                    \begin{cases}
                        \bm{r} & (\norm{\bm{r}}_2\leq R) \\
                        \frac{R^3}{\norm{\bm{r}}_2^3}\bm{r} & (\norm{\bm{r}}_2 > R)
                    \end{cases}
                \]
            \end{shadebox}
            \begin{proof}
                \quad\par
                必要なら系全体を回転させ，一般性を失わず$\bm{r} = [0,0,r]^\top\;(r>0)$とする。
                球座標変換を用いて積分を実行する。
                $\bm{r}' \coloneq r'[\sin\theta\cos\phi,\sin\theta\sin\phi,\cos\theta]^\top$とすると
                \begin{align*}
                    \bm{A}(\bm{r}) &= \integrate{0}{R}{\integrate{0}{\pi}{\integrate{0}{2\pi}{\frac{\bm{r}-\bm{r}'}{\norm{\bm{r}-\bm{r}'}_2^3}r'^2(\sin\theta)}{}{\phi}}{}{\theta}}{}{r'} \\
                    &= 2\pi\bm{i}_3\integrate{0}{R}{\integrate{0}{\pi}{\frac{r-r'\cos\theta}{(r^2+r'^2-2rr'\cos\theta)^{3/2}}r'^2(\sin\theta)}{}{\theta}}{}{r'}
                \end{align*}
                ここに$\bm{i}_3$は第3軸方向の単位ベクトルである。
                \par
                $f(r) \coloneq -(r^2+r'^2-2rr'\cos\theta)^{-1/2}r'^2\sin\theta$とすると次式が成り立つ。
                \[ \deriv{f}{r}{}(r) = \frac{(r-r'\cos\theta)r'^2\sin\theta}{(r^2+r'^2-2rr'\cos\theta)^{3/2}} \]
                上式右辺の$(r',\theta)=(0,0)$近傍での振る舞いを考える。
                $r'=r+\Delta r\;(\Delta r\neq 0)$とすると，$\sin\theta = O(\theta),\cos\theta = 1+O(\theta^2)$なので分母・分子ともに$r^3 O(\theta^3)$となる。
                よって積分領域に於いて$f$は$r$の$\mathrm{C}^1$級関数であるから積分と微分の順序交換が可能であり，次式が成り立つ。
                \begin{align*}
                    \bm{A}(\bm{r}) &= 2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\integrate{0}{\pi}{f(r)}{}{\theta}}{}{r'}}{r}{} \\
                    &= -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\integrate{0}{\pi}{\frac{r'^2\sin\theta}{\sqrt{r^2+r'^2-2rr'\cos\theta}}}{}{\theta}}{}{r'}}{r}{}
                \end{align*}
                $g(\theta) \coloneq \frac{r'}{r}\sqrt{r^2+r'^2-2rr'\cos\theta}$とすると次式が成り立つ。
                \[ \deriv{g}{\theta}{}(\theta) = \frac{r'^2\sin\theta}{\sqrt{r^2+r'^2-2rr'\cos\theta}} \]
                よって次式が成り立つ。
                \[ \bm{A}(\bm{r}) = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\left[g(\theta)\right]_0^\pi}{}{r'}}{r}{} = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\frac{r'}{r}\bigl(|r+r'|-|r-r'|\bigr)}{}{r'}}{r}{} \]
                $R<r$のときは次式が成り立つ。
                \[ \bm{A}(\bm{r}) = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{\frac{2r'^2}{r}}{}{r'}}{r}{} = \frac{4\pi R^3}{3r^2}\bm{i}_3 = \frac{4\pi R^3}{3}\frac{\bm{r}}{\norm{\bm{r}}_2^3} \]
                $R\geq r$のときは$r'$に関する積分領域を$[0,r]$と$[r,R]$に分けて考える。
                前者については$R<r$のときの結果で$R$を$r$で置き換えたものとなる。
                後者については$|r+r'|-|r-r'| = 2r$となり，次式が成り立つ。
                \[ \bm{A}(\bm{r}) = -2\pi\bm{i}_3\derivLong{\integrate{0}{R}{2r'}{}{r'}}{r}{} = \bm{0} \]
            \end{proof}

        \subsection{非負領域の共通部分の非負領域}
            \begin{shadebox}
                $\realNumbers^n$上で考える。$\bm{v}_1,\dotsc,\bm{v}_m\;(m\leq n)$は一次独立であるとし，$\bm{v}$の非負領域を
                \[P(\bm{v})\coloneq\left\{\bm{w}|\inprod{\bm{v}}{\bm{w}}\geq0\right\})\]
                で定義する。あるベクトル$\bm{u}$が$\bigcap_{i=1}^{m}{P(\bm{v}_i)}$の非負領域にある，すなわち
                \[\inprod{\bm{u}}{\bm{w}}\geq0,\forall\bm{w}\in\bigcap_{i=1}^{m}{P(\bm{v}_i)}\]
                であるとき，$\bm{u}$は適当な定数$a_i\geq0(i=1,\dotsc,m)$によって
                \[\bm{u} = \sum_{i=1}^{m}{a_i\bm{v}_i}\]

            \end{shadebox}

                \quad\par
                    まず$\bm{u}$が$\bm{v}_1,\dotsc,\bm{v}_m$の一次結合で表せることを示す。\\

                    \begin{equation}
                        \bm{u} = \sum_{i=1}^{m}{c_i\bm{v}_i} + \bm{w}_t
                    \end{equation}

                    \[\inprod{\bm{v}_i}{\bm{p}}\geq0,\forall\bm{v}_i\in W\]
                    また，$\bm{w}_{t2}$を$W^\bot$の任意のベクトルとすると
                    \[\inprod{\bm{v}_i}{\bm{w}_{t2}}=0,\forall\bm{v}_i\in W\]

                    \[\inprod{\bm{v}_i}{\bm{p}'}\geq0,\forall\bm{v}_i\in W\]

                                \vdots\\

                                \vdots\\

                            \end{array}
                        \right]

                            \quad\\

                            \quad

            線積分で表された定理の左辺を変形して面積分に置き換えて，定理の右辺と見比べる流れで行く。
            \par
            関数の引数を全部書いていてはスペースがいくらあっても足りないので省略して書くことになる(例えば$A_x(x,y,z)$を単に$A_x$と書いたり)のだが，こうすると独立変数と従属変数の関係を見誤る危険が生じるため注意が必要だ。
            迂闊に計算すると罠にはまる。
            特に危ない所では敢えて引数を明示する。
            \begin{align*}
                \oint_c{A\cdot\dvc} &= \oint_{\partial S}{\left(\vix A_x+\viy A_y+\viz A_z\right)\cdot\left(\vix \dx+\viy \dy+\viz \dz\right)} \\
                &= \oint_{\partial S}{\left(A_x\dx+A_y\dy\right)} + \oint_{\partial S}{A_z\dz} \\
                &= \underbrace{\oint_{\partial \textcolor{red}{\Omega}}{\left(A_x(x,y,S(x,y))\dx+A_y(x,y,S(x,y))\dy\right)}}_{\ctext{A}} + \underbrace{\oint_{\partial S}{A_z\dz}}_{\ctext{B}}
            \end{align*}
            \ctext{A}で積分経路が$\partial S$から$\partial\Omega$に変更されたことに注意。Greenの定理を使いたいから$xy$平面に落としたのだ。
            Greenの定理より
            \[
                \ctext{A} = \iint_\Omega{\left[\underbrace{\partDerivLong{A_y(x,y,S(x,y))}{x}{}}_{\ctext{C}}-\underbrace{\partDerivLong{A_x(x,y,S(x,y))}{y}{}}_{\ctext{D}}\right]\dx\dy}
            \]
            \ctext{C}が罠だ。引数を明示していないと$\left.\partDeriv{A_y}{x}{}\right|_{(x,y,z)=(x,y,S(x,y))}$と勘違いしてしまう。
            \newline
            \textcolor{red}{$\partDerivLong{A_y(x,y,S(x,y))}{x}{}$と$\left.\partDeriv{A_y}{x}{}\right|_{(x,y,z)=(x,y,S(x,y))}$は別物}であることに注意せなばならぬ。
            連鎖律に注意して
            \begin{align*}
                \ctext{C} &= \left.\partDeriv{A_y}{x}{}\right|_{(x,y,z)=(x,y,S(x,y))}\partDeriv{x}{x}{} + \left.\partDeriv{A_y}{y}{}\right|_{(x,y,z)=(x,y,S(x,y))}\partDeriv{y}{x}{} + \left.\partDeriv{A_y}{z}{}\right|_{(x,y,z)=(x,y,S(x,y))}\partDeriv{S}{x}{} \\
                &= \partDeriv{A_y}{x}{} + \partDeriv{A_y}{z}{}\partDeriv{S}{x}{}
            \end{align*}
            同様にして
            \[\ctext{D} = \partDeriv{A_x}{y}{} + \partDeriv{A_x}{z}{}\partDeriv{S}{y}{}\]
            よって結局
            \begin{align*}
                \ctext{A} &= \iint_\Omega{\left[\partDeriv{A_y}{x}{} + \partDeriv{A_y}{z}{}\partDeriv{S}{x}{} - \partDeriv{A_x}{y}{} - \partDeriv{A_x}{z}{}\partDeriv{S}{y}{}\right]\dx\dy} \\
                &= \iint_\Omega{\left[\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right) + \partDeriv{A_y}{z}{}\partDeriv{S}{x}{} - \partDeriv{A_x}{z}{}\partDeriv{S}{y}{}\right]\dx\dy}
            \end{align*}
            次に\ctext{B}を見る。$t$による偏微分を$\dot{}$で表すことにする。線積分の定義に従って
            \begin{align*}
                \ctext{B} &= \integrate{0}{1}{A_z(\phi(t),\psi(t),S(\phi(t),\psi(t))\left(\partDeriv{S}{x}{}\dot{\phi(t)}+\partDeriv{S}{y}{}\dot{\psi(t)}\right)}{}{t} \\
                &= \integrate{0}{1}{\left(A_z(\phi(t),\psi(t),S(\phi(t),\psi(t))\partDeriv{S}{x}{}\right)\dot{\phi(t)}}{}{t} + \integrate{0}{1}{\left(A_z(\phi(t),\psi(t),S(\phi(t),\psi(t))\partDeriv{S}{y}{}\right)\dot{\psi(t)}}{}{t} \\
                &= \oint_{\partial \Omega}{A_z(x,y,S(x,y))\partDeriv{S}{x}{}\dx} + \oint_{\partial \Omega}{A_z(x,y,S(x,y))\partDeriv{S}{y}{}\dy} \\
                &= \oint_{\partial \Omega}{\left(A_z(x,y,S(x,y))\partDeriv{S}{x}{}\dx + A_z(x,y,S(x,y))\partDeriv{S}{y}{}\dy\right)}
            \end{align*}
            これにGreenの定理を適用して，連鎖律に注意して偏微分していく。
            \begin{align*}
                \ctext{B} &= \iint_\Omega{\left[\partDerivLong{\left(A_z(x,y,S(x,y))\partDeriv{S}{y}{}\right)}{x}{} - \partDerivLong{\left(A_z(x,y,S(x,y))\partDeriv{S}{x}{}\right)}{y}{}\right]\dx\dy} \\
                &= \iint_\Omega{\left(\partDeriv{A_z}{x}{}\partDeriv{S}{y}{} + \partDeriv{A_z}{z}{}\partDeriv{S}{x}{}\partDeriv{S}{y}{} + A_z\partDerivIIHetero{S}{y}{x} - \partDeriv{A_z}{y}{}\partDeriv{S}{x}{} - \partDeriv{nA_z}{z}{}\partDeriv{S}{y}{}\partDeriv{S}{x}{} - A_z\partDerivIIHetero{S}{x}{y}\right)\dx\dy} \\
                &= \iint_\Omega{\left(\partDeriv{A_z}{x}{}\partDeriv{S}{y}{} - \partDeriv{A_z}{y}{}\partDeriv{S}{x}{}\right)\dx\dy}\quad\left(\because\;\text{$S$は$C^2$級と仮定したから$\partDerivIIHetero{S}{y}{x} = \partDerivIIHetero{S}{x}{y}$}\right)
            \end{align*}
            以上より
            \begin{align*}
                \oint_c{A\cdot\dvc} &= \iint_\Omega{\left[-\left(\partDeriv{A_z}{y}{}-\partDeriv{A_y}{z}{}\right)\partDeriv{S}{x}{}-\left(\partDeriv{A_x}{z}{}-\partDeriv{A_z}{x}{}\right)\partDeriv{S}{y}{}+\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right)\right]\dx\dy}
            \end{align*}
            \par
            これが定理の右辺と等しいことを示せば良い。
            まず$\dS$の正体を暴こう。感覚的には，$\dS$とは，$S$上で$x,y$が微小量$\dx,\dy$だけ動いた時に舐める$S$の面積をノルムに持ち，向きはその微小面上の任意の代表点での上向きの法線ベクトルに一致するようなベクトルであった。
            所謂「面積ベクトル」である。
            \par
            厳密にはRiemann和を取る段階で定義され，$S$上の点$(x,y,S(x,y))$における接平面のうち，その射影が$xy$平面上の矩形$\dx\dy$に収まる部分(3次元上の平行四辺形)の面積をノルムに持ち，向きは接平面の上向き法線ベクトルと一致するようなベクトルである。
            \[\dS = \left(\vix\dx+\viz\partDeriv{S}{x}{}\dx\right)\times\left(\viy\dy+\viz\partDeriv{S}{y}{}\dy\right) = \left(-\vix\partDeriv{S}{x}{} - \viy\partDeriv{S}{y}{} + \viz\right)\dx\dy\]
            よって定理の右辺は
            \begin{align*}
                \iint_S{\nabla\times A}\cdot\dS &= \iint_\Omega{\left[\vix\left(\partDeriv{A_z}{y}{}-\partDeriv{A_y}{z}{}\right)+\viy\left(\partDeriv{A_x}{z}{}-\partDeriv{A_z}{x}{}\right)+\viz\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right)\right]\cdot\dS} \\
                &= \iint_\Omega{\left[-\left(\partDeriv{A_z}{y}{}-\partDeriv{A_y}{z}{}\right)\partDeriv{S}{x}{}-\left(\partDeriv{A_x}{z}{}-\partDeriv{A_z}{x}{}\right)\partDeriv{S}{y}{}+\left(\partDeriv{A_y}{x}{}-\partDeriv{A_x}{y}{}\right)\right]\dx\dy} \\
                &= \oint_c{A\cdot\dvc}
            \end{align*}
        \end{proof}
