\chapter{Beta関数}
    \section{Beta関数とGamma関数の関係}
        \begin{shadebox}
            \[B(p,q) = \frac{\Gamma(p)\Gamma(q)}{\Gamma(p+q)}\]
        \end{shadebox}
        \begin{proof}
            \[\Gamma(p)\Gamma(q) = \parens*{\integrate{0}{\infty}{x^{p-1}\NapierE^{-x}}{}{x}}\parens*{\integrate{0}{\infty}{y^{q-1}\NapierE^{-y}}{}{y}} = \integrate{x=0}{\infty}{\integrate{y=0}{\infty}{x^{p-1}y^{q-1}\NapierE^{-x-y}}{}{y}}{}{x}\]
            積分経路を変更する。線分$(0,z)\to(z,0)$に沿った積分を$z:0 \to \infty$まで足し合わせる。上式の$y$は$z-x$に変わり
            \[\integrate{z=0}{\infty}{\parens*{\integrate{x=0}{z}{x^{p-1}(z-x)^{q-1}\NapierE^{-x+x-z}}{}{x}}}{}{z} = \integrate{z=0}{\infty}{\NapierE^{-z}\bracks*{\integrate{x=0}{z}{x^{p-1}(z-x)^{q-1}}{}{x}}}{}{z}\]
            (詳しく言えば，$x=u,y=v-u$なる変数変換を行ったのと等しい。ヤコビアンは1となる。)
            \\$x=uz$なる変数変換を行って
            \begin{align*}
                &\quad\integrate{z=0}{\infty}{\NapierE^{-z}\bracks*{\integrate{u=0}{1}{z^{p-1}u^{p-1}z^{q-1}(1-u)^{q-1} z}{}{u}}}{}{z} = \integrate{z=0}{\infty}{\NapierE^{-z}z^{p+q-1}\bracks*{\integrate{u=0}{1}{u^{p-1}(1-u)^{q-1}}{}{u}}}{}{z}\\
                &= \integrate{0}{\infty}{\NapierE^{-z}z^{p+q-1}B(p,q)}{}{z} = B(p,q)\integrate{0}{\infty}{\NapierE^{-z}z^{p+q-1}}{}{z} = B(p,q)\Gamma(p+q)
            \end{align*}
        \end{proof}
    \section{多変量Beta関数}
        \begin{shadebox}
            $ \bm{\alpha} = [\alpha_1,\dotsc,\alpha_n]^\top > \bm{0}$に対して\textbf{多変量Beta関数}を次の広義積分で定義する。
            \begin{align*}
                B(\bm{\alpha}) &\coloneq \integrate{\Omega}{}{\prod_{i=1}^{n-1} x_i^{\alpha_i-1} \parens*{1-\sum_{i=1}^{n-1} x_i}^{\alpha_n-1}}{}{\Omega}, \quad \Omega \coloneq \setComprehension*{[x_1,\dotsc,x_{n-1}]^\top}{x_i\geq 0, \sum_{i=1}^{n-1} x_i\leq 1} \\
                &= \integrate{x_1=0}{1}{\integrate{x_2=0}{1-x_1}{\dotsi\integrate{x_{n-1}=0}{1-\sum_{i=1}^{n-1} x_i}{\prod_{i=1}^{n-1} x_i^{\alpha_i-1}\parens*{1-\sum_{i=1}^{n-2} x_i}^{\alpha_n-1}}{}{x_{n-1}}\dotsi}{}{x_2}}{}{x_1}
            \end{align*}
        \end{shadebox}
    \section{多変量Beta関数とGamma関数の関係}
        \begin{shadebox}
            \[B(\bm{\alpha}) = \frac{\prod_{i=1}^n\Gamma(\alpha_i)}{\Gamma(\sum_{i=1}^n \alpha_i)}\]
        \end{shadebox}
        \begin{proof}
            \[ \prod_{i=1}^n \Gamma(\alpha_i) = \prod_{i=1}^n \integrate{0}{\infty}{x_i^{\alpha_i-1}\NapierE^{-x_i}}{}{x_i} = \integrate{x_1=0}{\infty}{\dotsi\integrate{x_n=0}{\infty}{\prod_{i=1}^n x_i^{\alpha_i-1}\exp\parens*{-\sum_{i=1}^n x_i}}{}{x_n}\dotsi}{}{x_1} \]
            ここで
            \[
                \vy = P\vx,\quad
                P \coloneq
                \begin{bmatrix}
                    \bm{e}_1^\top \\
                    \bm{e}_2^\top \\
                    \vdots \\
                    \bm{e}_{n-1}^\top \\
                    \bm{1}_n^\top
                \end{bmatrix}
            \]
            なる変数変換を行うと，$x_n = y_n - \sum_{i=1}^{n-1}x_i = y_n - \sum_{i=1}^{n-1}y_i,|P|=1$なので
            \begin{align*}
                \prod_{i=1}^n \Gamma(\alpha_i) &= \integrate{y_n=0}{\infty}{\integrate{y_1=0}{y_n}{\dotsi\integrate{y_{n-1}=0}{y_n-\sum_{i=1}^{n-2}y_i}{\prod_{i=1}^{n-1}y_i^{\alpha_i-1}\parens*{y_n-\sum_{i=1}^{n-1} y_i }^{\alpha_n-1} \NapierE^{-y_n}|P|^{-1}}{}{y_{n-1}}\dotsi}{}{y_1}}{}{y_n} \\
                &= \integrate{y_n=0}{\infty}{\NapierE^{-y_n}\integrate{y_1=0}{y_n}{\dotsi\integrate{y_{n-1}=0}{y_n-\sum_{i=1}^{n-2}y_i}{\prod_{i=1}^{n-1}y_i^{\alpha_i-1}\parens*{y_n-\sum_{i=1}^{n-1} y_i }^{\alpha_n-1}}{}{y_{n-1}}\dotsi}{}{y_1}}{}{y_n}
            \end{align*}
            ここで$y_i = z_iy_n\;(i=1,\dotsc,n-1)$と変数変換すると
            \begin{align*}
                \prod_{i=1}^n \Gamma(\alpha_i) &= \integrate{y_n=0}{\infty}{y_n^{\parens*{\sum_{i=1}^n \alpha_i-1}}\NapierE^{-y_n}}{}{y_n}\integrate{z_1=0}{1}{\dotsi\integrate{z_{n-1}=0}{1-\sum_{i=1}^{n-2}z_i}{\prod_{i=1}^{n-1}z_i^{\alpha_i-1}\parens*{1-\sum_{i=1}^{n-1} z_i }^{\alpha_n-1}}{}{z_{n-1}}\dotsi}{}{z_1} \\
                &= \Gamma\parens*{\sum_{i=1}^n \alpha_i}B(\bm{\alpha})
            \end{align*}
        \end{proof}
