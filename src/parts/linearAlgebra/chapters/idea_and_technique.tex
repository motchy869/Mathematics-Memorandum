    \section{構成}
        \subsection{冪等行列の構成例}
            $N\in\naturalNumbers,n\leq N,\vv_1,\dots,\vv_n\in\complexNumbers^N$とし，$\vv_1,\dots,\vv_n$は互いに直交し，ノルムが1であるとする。
            このとき$A \coloneq \sum_{i=1}^n \vv_i\HerConj{\vv_i}$は$A^n = I\;(n=1,2,3,\dots)$を満たす。
    \section{演算}
        \subsection{積和の階数を拡張する}
            以下，$\field$は適当な体を，$l,m,n$は適当な自然数を表す。
            \begin{enumerate}
                \item スカラー同士の積: $x,y\in\field$に対して$xy$
                \item 内積(スカラー同士の積の和): $\vx,\vy \in \field^l$に対して$\vx^\top\vy = \sum_{i=1}^l x_i y_i$
                \item トレース(内積同士の和): $X = [\vx_1,\dotsc,\vx_n]^\top,Y = [\vy_1,\dotsc,\vy_n]^\top\;(\vx_1,\vx_2,\dotsc,\vy_n \in \field^l)$に対して$\tr{X^\top Y} = \sum_{i=1}^l \vx_i^\top\vy_i$
            \end{enumerate}
