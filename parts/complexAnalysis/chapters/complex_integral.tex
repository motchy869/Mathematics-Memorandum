\chapter{複素積分}
    \section{無限に大きい半円周上での\texorpdfstring{$\NapierE^{iz}/z$}{e\string^(iz)/z}の積分}
        \label{無限に大きい半円周上でのe^(iz)/zの積分}
        \begin{shadebox}
            $R>0$とし、パラメータ表示された曲線$C:\;z = R\NapierE^{i\theta};\;\theta :0\to\pi$を考える。
            このとき次式が成り立つ。
            \[ \abs{\integrate{C}{}{\frac{\NapierE^{iz}}{z}}{}{z}} < \frac{\pi}{R}\left(1-\NapierE^{-R}\right) \]
        \end{shadebox}
        \begin{proof}
            \quad\par
            $\mathrm{d}z = iR\NapierE^{i\theta}\mathrm{d}\theta$であるから
            \begin{align*}
                S &:= \integrate{C}{}{\frac{\NapierE^{iz}}{z}}{}{z} = \integrate{0}{\pi}{\frac{1}{R\NapierE^{i\theta}}\exp iR(\cos\theta + i\sin\theta) iR\NapierE^{i\theta}}{}{\theta} = i\integrate{0}{\pi}{\exp (iR\cos\theta - R\sin\theta)}{}{\theta} \\
                |S| &= \abs{\integrate{0}{\pi}{\exp (iR\cos\theta - R\sin\theta)}{}{\theta}} \leq \integrate{0}{\pi}{\abs{\exp (iR\cos\theta - R\sin\theta)}}{}{\theta} \\
                &= \integrate{0}{\pi}{\NapierE^{-R\sin\theta}}{}{\theta} = 2\integrate{0}{\pi/2}{\NapierE^{-R\sin\theta}}{}{\theta}
            \end{align*}
            $2\theta/\pi < \sin\theta\;(\theta\in(0,\pi/2))$であるから次式が成り立つ。
            \[ 2\integrate{0}{\pi/2}{\NapierE^{-R\sin\theta}}{}{\theta} < 2\integrate{0}{\pi/2}{\NapierE^{-2R\theta/\pi}}{}{\theta} = \frac{\pi}{R}\left(1-\NapierE^{-R}\right) \]
        \end{proof}
    \section{例題}
        \subsection{\texorpdfstring{$\integrate{0}{2\pi}{\Log(1-a\exp(i\theta))}{}{\theta}=0\quad (C\text{: 原点中心の単位円},\;a\in\realNumbers)$}{Log(1-a exp(iΘ))の単位円周上での積分は0}}
            \begin{solution}
                \quad\par
                $a=0$のときは明らかに成り立つ。$a\neq0$のときは$z=\NapierE^{i\theta}$とすると
                \[\integrate{0}{2\pi}{\Log(1-a\NapierE^{i\theta})}{}{\theta} = \integrate{C}{}{\frac{\Log(1-az)}{iz}}{}{z}\quad (C:\text{原点中心の左回り単位円})\]
                被積分関数の$C$内部の特異点は$z=0$である。原点の近傍(具体的には$|z|<1/|a|$の範囲)においては
                \[\frac{\Log(1-az)}{iz} = \frac{-1}{iz}\sum_{n=1}^\infty\frac{(az)^n}{n}=\frac{-1}{i}\sum_{n=1}^\infty\frac{a^nz^{n-1}}{n} = \frac{-1}{i}\sum_{n=0}^\infty\frac{a^{n+1}}{n+1}z^n\]
                であり、特異点$z=0$は除去可能であることから従う。
            \end{solution}
