\chapter{表記}
    \begin{itemize}
        \item $\realNumbers_+ \coloneq \{x\in\realNumbers|x\geq 0\}$ : 非負の実数全体の集合
        \item $\realNumbers_{++} \coloneq \{x\in\realNumbers|x > 0\}$ : 正の実数全体の集合
        \item $f(x) = \LittleO{x\to a}{x}$ : $x\to a$ の極限で$x$が$a$に十分近いとき、 $f(x)$ は $x$ に比べて十分速く0に収束する
        \item $\bm{1}_n$ : 要素が全て1である$n$次元列ベクトル
        \item $\bm{0}_n$ : 要素が全て0である$n$次元列ベクトル
        \item $\bm{x}[i]$ : 列(行)ベクトルの第$i$成分
        \item $\matEntry{A}{i}{j},\;A_{i,j}$ : 行列$A$の第$(i,j)$成分
        \item $\matPart{A}{i_1}{i_2}{j_1}{j_2}$ : 行列$A$の第$i_1$行から第$i_2$行までの部分と第$j_1$列から第$j_2$列までの部分を抽出したもの
        \item $\matEntry{A}{i}{:}$ : 行列$A$の第$i$行
        \item $\matEntry{A}{:}{j}$ : 行列$A$の第$j$列
        \item $[A]_{i,j}$ : 行列$A$の第$i$行$j$列を取り除いてできる部分行列。
    \end{itemize}
