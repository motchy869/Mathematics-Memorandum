    \section{変数変換 : \texorpdfstring{$\integrate{x(t_0)}{x(t)}{f(x)}{}{x} = \integrate{t_0}{t}{f(x)x'(t)}{}{t}$}{∫\_x(t0)\string^x(t) f(x)dx = ∫\_t0\string^t f(x)x'(t)dt}}
        \begin{shadebox}
            $t$の関数$x(t)$とその関数$f(x)$について
            \[\integrate{x(t_0)}{x(t)}{f(x)}{}{x} = \integrate{t_0}{t}{f(x)x'(t)}{}{t}\]
        \end{shadebox}
        \begin{proof}
            \quad\par
            $t=0$について
            \[\integrate{x(t_0)}{x(t)}{f(x)}{}{x} = \integrate{x(t_0)}{x(t_0)}{f(x)}{}{x} = 0\]
            \[\integrate{t_0}{t}{f(x)x'(t)}{}{t} = \integrate{t_0}{t_0}{f(x)x'(t)}{}{t} = 0\]
            より両者は一致する。
            次に
            \[\derivLong{\integrate{x(t_0)}{x(t)}{f(x)}{}{x}}{t}{} = \left(\derivLong{\integrate{x(t_0)}{x(t)}{f(x)}{}{x}}{x}{}\right)\deriv{x}{t}{} = f(x)x'(t) = \derivLong{\integrate{t_0}{t}{f(x)x'(t)}{}{t}}{t}{}\]
            より両者の微分係数は一致する。
            以上より両者は等しい。
        \end{proof}
