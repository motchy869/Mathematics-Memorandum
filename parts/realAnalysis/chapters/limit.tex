\chapter{極限}
    \section{問題例}
        \subsection{\texorpdfstring{$\lim_{x\to\infty} \sqrt{x^2+1} - x = 0$}{√(x\string^2+1)-x → 0 as x→∞}}
            \begin{proof}
                \quad\par
                $y(x) := \sqrt{x^2+1} - x$とおく。
                $y(x) > 0$は明らか。
                \[ y(x)+x = \sqrt{x^2+1} \quad \therefore y(x)^2 + 2xy(x) + x^2 = x^2 + 1 \quad \therefore y(x)^2 + 2xy(x) = 1 \]
                \[ \therefore y(x)\bigl(y(x) + 2x\bigr) = 1 \quad \therefore 0 < y(x) = \frac{1}{2x + y(x)} < \frac{1}{2x} \to 0\;\text{as}\;x\to\infty \]
                挟みうちの原理から$y(x) \to 0$である。
            \end{proof}
        \subsection{\texorpdfstring{$a>0,\;\lim_{x\to+0}\frac{x^2}{\sqrt{x^2+a^2}-a} = 2a$}{x\string^2/(√(x\string^2+a\string^2)-a) → 2a as x → +0}}
            \begin{proof}
                \quad\par
                $y = x + \sqrt{x^2+a^2} - a$とおき、$x$について解くと$x = \frac{y(y+2a)}{2(y+a)}$となる。
                また、$\sqrt{x^2+a^2}-a = y - x = \frac{y^2}{2(y+a)}$となる。
                $x\to+0$のとき$y\to+0$であり、
                \[ \lim_{x\to+0}\frac{x^2}{\sqrt{x^2+a^2}-a} = \lim_{y\to+0} \frac{(y+2a)^2}{2(y+a)} = 4a^2/(2a) = 2a \]
            \end{proof}
