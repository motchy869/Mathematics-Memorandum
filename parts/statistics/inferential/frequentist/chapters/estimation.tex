\chapter{推定}
    \section{標本分散}
        \subsection{\texorpdfstring{$\E{}{\frac{1}{n}\sum_{i=1}^{n}{(X_i-\overline{X})^2}} = \frac{n-1}{n}\sigma^2$}{E((1/n)Σ\_\{i=1\}\string^n (Xi - sampleAvg(X))\string^2) = ((n-1)/n)σ\string^2}}
            \begin{shadebox}
                $X$が平均$\mu$,分散$\sigma^2$の母集団に属しているとき、そこからランダムに抽出した標本$(X_1,X_2,\dots,X_n)$の標本分散$\displaystyle S^2 \coloneqq \frac{1}{n}\sum_{i=1}^{n}{(X_i-\overline{X})^2}$の期待値は$\displaystyle \text{E}(S^2) = \frac{n-1}{n}\sigma^2$である。
            \end{shadebox}
            \begin{proof}
            {%スコープ。\s 上書き回避
                \newcommand{\EE}[1]{\text{E}^2\left[#1\right]}
                \renewcommand{\s}{\sigma^2}
                \newcommand{\mX}{\overline{X}}
                \begin{equation}
                    \E{}{S^2} = \E{}{\frac{1}{n}\sum_{i=1}^{n}{(X_i-\mX)^2}} = \frac{1}{n}\sum_{i=1}^{n}{\E{}{(X_i-\mX)^2}} = \frac{1}{n}\sum_{i=1}^{n}{\left(\E{}{X_i^2}+\E{}{\mX^2}-2\E{}{\mX X_i}\right)}
                \end{equation}
                然るに\\
                $\Var{}{X_i} = \E{}{X_i^2}-\EE{X_i}$より
                \begin{equation}
                    \E{}{X_i^2} = \Var{}{X_i}+\EE{X_i} = \s+\mu^2
                \end{equation}
                \begin{align}
                    \E{}{\mX^2} &= \frac{1}{n^2}\E{}{\left(\sum_{i=1}^{n}{X_i}\right)^2} = \frac{1}{n^2}\E{}{\sum_{i=1}^{n}{X_i^2} + \sum_{i\neq j}^{}{X_iX_j}} = \frac{1}{n^2}\sum_{i=1}^{n}{\E{}{X_i^2}} + \frac{1}{n^2}\sum_{i\neq j}^{}{\E{}{X_iX_j}} \nonumber \\
                    &= \frac{1}{n^2}\sum_{i=1}^{n}{(\s+\mu^2)} + \frac{1}{n^2}\sum_{i\neq j}^{}{\E{}{X_i}\E{}{X_j}} = \frac{1}{n^2}\sum_{i=1}^{n}{(\s+\mu^2)} + \frac{1}{n^2}\sum_{i\neq j}^{}{\mu^2} \nonumber \\
                    &= \frac{1}{n}(\s+\mu^2) + \frac{1}{n^2}(n^2-n)\mu^2 = \frac{1}{n}\s + \mu^2
                \end{align}
                \begin{align}
                    \E{}{\mX X_i} &= \frac{1}{n}\E{}{(X_1+X_2+\dotsb+X_n)X_i} = \frac{1}{n}\E{}{X_i^2 + \sum_{j=1,\neq i}^{n}{X_jX_i}} \nonumber \\
                    &= \frac{1}{n}\E{}{X_i^2} + \frac{1}{n}\sum_{j=1,\neq i}^{n}{\E{}{X_jX_i}} = \frac{1}{n}(\s+\mu^2) + \frac{1}{n}\sum_{j=1,\neq i}^{n}{\E{}{X_j}\E{}{X_i}} \nonumber \\
                    &= \frac{1}{n}(\s+\mu^2) + \frac{1}{n}\sum_{j=1,\neq i}^{n}{\mu^2} = \frac{1}{n}(\s+\mu^2) + \frac{1}{n}(n-1)\mu^2 = \frac{1}{n}\s + \mu^2
                \end{align}
                (2),(3),(4)を(1)に適用して
                \[\E{}{\s} = \frac{1}{n}\sum_{i=1}^{n}{\left(\s-\frac{1}{n}\s\right)} = \frac{n-1}{n}\s\]
            }
            \end{proof}
