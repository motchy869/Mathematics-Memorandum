\chapter{beta分布}
    \section{Beta分布に従う確率変数の生成}
        \subsection{\texorpdfstring{$X\sim\GammaDist{\alpha}{1},Y\sim\GammaDist{\beta}{1}\Rightarrow \frac{X}{X+Y}\sim\Beta{\alpha}{\beta}$}{X～Gamma(α,1), Y～Gamma(β,1) ⇒ X/(X+Y)～Beta(α,β)}}
            \begin{proof}
                \quad\par
                $Z=\frac{X}{X+Y}$とおく。
                \begin{align*}
                    \PDF{Z}{z} &= \partDerivLong{\Pr{Z\leq z}}{z}{} = \partDerivLong{\Pr{X\leq\frac{z}{1-z}Y}}{z}{} \\
                    &= \partDerivLong{\integrate{y=0}{\infty}{\integrate{x=0}{\frac{z}{1-z}y}{ \PDF{\GammaDist{\alpha}{1}}{x}\PDF{\GammaDist{\beta}{1}}{y} }{}{x}}{}{y}}{z}{} \\
                    &= \partDerivLong{\integrate{y=0}{\infty}{\integrate{x=0}{\frac{z}{1-z}y}{\frac{x^{\alpha-1}e^{-x}}{\Gamma(\alpha)}\frac{y^{\beta-1}e^{-y}}{\Gamma(\beta)}}{}{x}}{}{y}}{z}{} \\
                    &= \frac{1}{\Gamma(\alpha)\Gamma(\beta)}\integrate{y=0}{\infty}{\partDerivLong{\integrate{x=0}{\frac{z}{1-z}y}{x^{\alpha-1}e^{-x}y^{\beta-1}e^{-y}}{}{x}}{z}{}}{}{y} \\
                    &= \frac{1}{\Gamma(\alpha)\Gamma(\beta)}\integrate{y=0}{\infty}{y^{\beta-1}e^{-y}\left(\frac{z}{1-z}y\right)^{\alpha-1}\exp\left(-\frac{z}{1-z}y\right)\left(\partDerivLong{\frac{z}{1-z}y}{z}{}\right)}{}{y} \\
                    &= \frac{1}{\Gamma(\alpha)\Gamma(\beta)}\integrate{y=0}{\infty}{y^{\beta-1}\exp\left(\frac{-y}{1-z}y\right)\left(\frac{y}{1-z}\right)^{\alpha-1}z^{\alpha-1}\frac{1}{(1-z)^2}}{}{y} \\
                    &= \frac{z^{\alpha-1}(1-z)^{\beta-1}}{\Gamma(\alpha)\Gamma(\beta)}\integrate{0}{\infty}{u^{\alpha+\beta-1}e^{-u}}{}{u} \quad\left(u=\frac{y}{1-z}\textrm{とおいた}\right) \\
                    &= \frac{\Gamma(\alpha+\beta)}{\Gamma(\alpha)\Gamma(\beta)}z^{\alpha-1}(1-z)^{\beta-1} = \frac{z^{\alpha-1}(1-z)^{\beta-1}}{B(\alpha,\beta)} = \PDF{\Beta{\alpha}{\beta}}{z}
                \end{align*}
            \end{proof}
