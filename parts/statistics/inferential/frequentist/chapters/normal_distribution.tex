    \section{裾確率の評価}
        \begin{shadebox}
            標準正規分布の累積分布関数\texorpdfstring{$\Phi(x)$}{Φ(x)}に対して次が成り立つ。
            \[1-\Phi(\alpha) \leq \frac{1}{2}\exp{\left(-\frac{1}{2}\alpha^2\right) \quad \textrm{for }^\forall \alpha \geq 0}\]
        \end{shadebox}
        \begin{proof}
            標準正規分布の確率密度関数を$\phi(\alpha)$とすると、任意の$\alpha>0$に対して次式が成り立つ。
            \[1-\Phi(\alpha) = \integrate{\alpha}{\infty}{\phi(x)}{}{x} \leq \integrate{\alpha}{\infty}{\frac{x}{\alpha}\phi(x)}{}{x} = \frac{1}{\alpha}\left[-\phi(x)\right]_\alpha^\infty = \frac{1}{\alpha}\phi(\alpha)\]
            よって$\alpha\geq 1$に対しては
            \[1-\Phi(\alpha) \leq \phi(\alpha) = \frac{1}{\sqrt{2\pi}}e^{-\frac{1}{2}\alpha^2} \leq \frac{1}{2}e^{-\frac{1}{2}\alpha^2}\]
            $\alpha<1$に対しては別のやり方で示す。$1-\Phi(\alpha) = 1/2 - \integrate{0}{\alpha}{\phi(x)}{}{x} =: 1/2-S(\alpha)$であり、$\phi(x)$の変曲点は$x=\pm 1$である。
            \begin{figure}[H]
                \centering
                \includegraphics[keepaspectratio, scale=0.5]
                {\currfiledir/imgs/evaluation-of-tail-weight-of-gaussian-distribution/graph.eps}
                \caption{$\phi(x)$のグラフ}
            \end{figure}
            \begin{align*}
                S(\alpha) &\geq \textrm{赤い台形の面積} = \frac{\alpha}{2\sqrt{2\pi}}(1+e^{-\frac{1}{2}\alpha^2}) \\
                \therefore\;1-\Phi(\alpha) &\leq \frac{1}{2}\left(1-\frac{\alpha}{\sqrt{2\pi}}(1+e^{-\frac{1}{2}\alpha^2})\right) \\
                \therefore\;1-\Phi(\alpha)-\frac{1}{2}e^{-\frac{1}{2}\alpha^2} &\leq \frac{1}{2}\left(1-\frac{\alpha}{\sqrt{2\pi}}(1+e^{-\frac{1}{2}\alpha^2})-e^{-\frac{1}{2}\alpha^2}\right) =: g(\alpha)/2
            \end{align*}
            $g(\alpha)$は区間$[0,1]$で凸であり(凸関数の和が凸関数であることと、2階導関数の計算によりわかる)、$g(0)=0$である。
            また、数値計算によると$g(1)\approx 1-\frac{4.8}{4.2}<0$であるから区間$[0,1]$で$g(\alpha)\leq 0$
        \end{proof}
    \section{スケール変換とシフト}
        \begin{shadebox}
            \[X \sim N(\mu,\sigma^2) \Leftrightarrow Z=aX+b \sim N(a\mu+b,(a\sigma)^2) \hspace{10pt} (a \neq 0)\]
        \end{shadebox}
        \begin{proof}
            \quad\par
            $(\Rightarrow)$\newline
            $X$の確率密度関数を$f^X(x)$と書くことにすると
            \[f^X(x) = \frac{1}{\sqrt{2\pi}\sigma}\exp\left[-\frac{(x-\mu)^2}{2\sigma^2}\right]\]
            (i) $a>0$の場合\par
            \[P(-\infty \leq Z \leq z) = P\left(\-\infty \leq X \leq \frac{z-b}{a}\right) = \int_{-\infty}^{\frac{z-b}{a}}{f^X(x) dx}\]
            \[\therefore \; f^Z(z) = \frac{d}{dz}P(-\infty \leq Z \leq z) = f^X(\frac{z-b}{a})\cdot\frac{d}{dz}\left(\frac{z-b}{a}\right) = \frac{1}{\sqrt{2\pi}(a\sigma)}\exp\left[-\frac{(z-a\mu-b)^2}{2(a\sigma)^2}\right]\]
            \[\therefore f^Z(z) = \frac{1}{\sqrt{2\pi}(a\sigma)}\exp\left[-\frac{(z-a\mu-b)^2}{2(a\sigma)^2}\right] = \text{pdf of  }N(a\mu+b,(a\sigma)^2)\]
            (ii) $a<0$の場合\par
            \[P(-\infty \leq Z \leq z) = P\left(\frac{z-b}{a} \leq X \leq \infty\right) = \int_{\frac{z-b}{a}}^{\infty}{f(x) dx}\]
            \begin{align*}
                \therefore \; f^Z(z) &=& \frac{d}{dz}P(-\infty \leq Z \leq z) = \textcolor{red}{-}f^X(\frac{z-b}{a})\cdot\frac{d}{dz}\left(\frac{z-b}{a}\right) = \frac{\textcolor{red}{-1}}{\sqrt{2\pi}(a\sigma)}\exp\left[-\frac{(z-a\mu-b)^2}{2(a\sigma)^2}\right]\\
                &=& \frac{1}{\sqrt{2\pi}(\textcolor{red}{|a|}\sigma)}\exp\left[-\frac{(z-a\mu-b)^2}{2(a\sigma)^2}\right] = \text{pdf of  }N(a\mu+b,(a\sigma)^2)
            \end{align*}
            \newline
            $(\Leftarrow)$\newline
            $Y \sim N(a\mu+b,(b\sigma)^2)$として、$(Y-b)/a\;(=X)$の分布を既に示された必要性に依って計算すれば明らかである。
        \end{proof}
    \section{再生性}
        \begin{shadebox}
            \[X \sim N(\mu_X,{\sigma_X}^2)\;\;,\;\;Y \sim N(\mu_Y,{\sigma_Y}^2) \quad \Rightarrow \quad Z=X+Y \sim N(\mu_X+\mu_Y,{\sigma_X}^2+{\sigma_Y}^2)\]
        \end{shadebox}
        \begin{proof}
            \quad\par
            $X,Y,Z$の確率密度関数を$f^X(x),f^Y(y),f^Z(z)$と書くことにすると
            \begin{align*}
                f^Z(z) &= \derivLong{\integrate{x=-\infty}{\infty}{\integrate{y=-\infty}{z-x}{f^X(x)f^Y(y)}{}{y}}{}{x}}{z}{} = \int_{-\infty}^{\infty}{f^X(x)f^Y(-x+z) dx} \quad(\because\ref{重積分の微分の系}) \\
                &= \int_{-\infty}^{\infty}{\frac{1}{\sqrt{2\pi}\sigma_X}\exp{\left[-\frac{(x-\mu_X)^2}{2{\sigma_X}^2}\right]}\frac{1}{\sqrt{2\pi}\sigma_Y}\exp{\left[-\frac{(-x+z-\mu_Y)^2}{2{\sigma_Y}^2}\right]} dx}\\
                &= \frac{1}{2\pi\sigma_X\sigma_Y}\int_{-\infty}^{\infty}{\exp{\left[-\frac{(x-\mu_X)^2}{2{\sigma_X}^2} - \frac{(-x+z-\mu_Y)^2}{2{\sigma_Y}^2}\right]} dx}
            \end{align*}
            expの引数を$A$とおくと
            \begin{align*}
                A &= -\frac{1}{{2\sigma_X}^2{\sigma_Y}^2}\left[{\sigma_Y}^2(x-\mu_X)^2 + {\sigma_X}^2(-x+z-\mu_Y)^2\right]\\
                &= -\frac{1}{{2\sigma_X}^2{\sigma_Y}^2}\left[{\sigma_X}^2(x^2+{\mu_X}^2-2\mu_Xx)+{\sigma_X}^2(x^2+z^2+{\mu_Y}^2-2zx-2z\mu_Y+2\mu_yx)\right]\\
                &= -\frac{1}{{2\sigma_X}^2{\sigma_Y}^2}\left[({\sigma_X}^2+{\sigma_Y}^2)x^2 + (-2{\sigma_Y}^2\mu_X - 2{\sigma_X}^2z + 2{\sigma_X}^2\mu_Y)x \right.\\
                &\quad \left. + {\sigma_Y}^2{\mu_X}^2 + {\sigma_X}^2z^2 + {\sigma_X}^2{\mu_Y}^2 - 2{\sigma_X}^2z\mu_Y\right]\\
                &= -\frac{1}{{2\sigma_X}^2{\sigma_Y}^2}\left\{{\left[\sqrt{{\sigma_X}^2+{\sigma_Y}^2}x - \frac{{\sigma_Y}^2\mu_X + {\sigma_X}^2z - {\sigma_X}^2\mu_Y}{\sqrt{{\sigma_X}^2+{\sigma_Y}^2}}\right]}^2\right.\\
                &\quad -\frac{1}{{\sigma_X}^2+{\sigma_Y}^2}\left({\sigma_Y}^4{\mu_X}^2 + {\sigma_X}^4z^2 + {\sigma_X}^4{\mu_Y}^2 + 2{\sigma_X}^2{\sigma_Y}^2\mu_X z - 2{\sigma_X}^4\mu_Y z - 2{\sigma_X}^2{\sigma_Y}^2\mu_X\mu_Y\right)\\
                &\quad \left. + {\sigma_Y}^2{\mu_X}^2 + {\sigma_X}^2z^2 + {\sigma_X}^2{\mu_Y}^2 - 2{\sigma_X}^2z\mu_Y \right\}\\
                &= -\frac{1}{{2\sigma_X}^2{\sigma_Y}^2} \left\{{\left[\sqrt{{\sigma_X}^2+{\sigma_Y}^2}x - \frac{{\sigma_Y}^2\mu_X + {\sigma_X}^2z - {\sigma_X}^2\mu_Y}{\sqrt{{\sigma_X}^2+{\sigma_Y}^2}}\right]}^2\right.\\
                &\quad \left. \frac{{\sigma_X}^2{\sigma_Y}^2}{{\sigma_X}^2+{\sigma_Y}^2}({\mu_X}^2 + z^2 + {\mu_Y}^2 - 2\mu_Y z - 2\mu_X z + 2\mu_X\mu_Y) \right\}\\
                &= -\frac{1}{{2\sigma_X}^2{\sigma_Y}^2} \left\{{\left[\sqrt{{\sigma_X}^2+{\sigma_Y}^2}x - \frac{{\sigma_Y}^2\mu_X + {\sigma_X}^2z - {\sigma_X}^2\mu_Y}{\sqrt{{\sigma_X}^2+{\sigma_Y}^2}}\right]}^2 + \frac{{\sigma_X}^2{\sigma_Y}^2}{{\sigma_X}^2+{\sigma_Y}^2}(z-\mu_X-\mu_Y)^2 \right\}
            \end{align*}
            これと(1)より
            \begin{align*}
                f^z(z) &= \frac{1}{2\pi\sigma_X\sigma_Y}\exp{\left[-\frac{(z-\mu_X-\mu_Y)^2}{2({\sigma_X}^2+{\sigma_Y}^2)}\right]}\int_{-\infty}^{\infty}{\exp{\left[\frac{{\sigma_X}^2+{\sigma_Y}^2}{2{\sigma_X}^2{\sigma_Y}^2}\left(x-\frac{{\sigma_Y}^2\mu_X + {\sigma_X}^2z - {\sigma_X}^2\mu_Y}{{\sigma_X}^2+{\sigma_Y}^2}\right)^2\right]} dx}\\
                &= \frac{1}{\sqrt{2\pi}\sqrt{{\sigma_X}^2+{\sigma_Y}^2}}\exp{\left[-\frac{(z-\mu_X-\mu_Y)^2}{2({\sigma_X}^2+{\sigma_Y}^2)}\right]} = \text{pdf of} \quad N(\mu_X+\mu_Y,{\sigma_X}^2+{\sigma_Y}^2)
            \end{align*}
        \end{proof}

    \section{分散\texorpdfstring{$\sigma^2$}{σ\string^2},平均\texorpdfstring{$\mu$}{μ}のときの累積分布関数は\texorpdfstring{$\Phi(\frac{z-\mu}{\sigma})$}{Φ((z-μ)/σ)}}
        \begin{shadebox}
            標準正規分布の累積分布関数を$\Phi(z) \coloneqq \integrate{-\infty}{z}{\phi(x)}{}{x}$とするとき、
            $X\sim N(\mu,\sigma)$の累積分布関数は$\Phi_X(Z) = \Phi(\frac{z-\mu}{\sigma})$となる。
        \end{shadebox}
        \begin{proof}
            \begin{align*}
                \Phi_X(z) &= \integrate{-\infty}{z}{\frac{1}{\sigma}\func{\phi}{\frac{z-\mu}{\sigma}}}{}{x} = \integrate{-\infty}{\frac{z-\mu}{\sigma}}{}{\phi(y)}{y}\quad\left(\frac{z-\mu}{\sigma}=y\textrm{とおいた}\right) \\
                &= \func{\Phi}{\frac{z-\mu}{\sigma}}
            \end{align*}
        \end{proof}
    \section{零平均,同分散の正規分布に独立に従う\texorpdfstring{$n$}{n}個の確率変数の直交変換もまた同じ分布に従う}
        \label{零平均,同分散の正規分布に独立に従うn個の確率変数の直交変換もまた同じ分布に従う}
        \begin{shadebox}
            $X_1,\dotsc,X_n\sim N(0,\sigma^2),\quad\bm{X}\coloneqq[X_1,\dotsc,X_n]^\top$とし、$P\in\realNumbers^{n\times n}$を直交行列とする。
            $\bm{Y} \coloneqq [Y_1,\dotsc,Y_n]^\top \coloneqq P\bm{X}$とすると、$Y_1,\dotsc,Y_n\sim N(0,\sigma^2)$
        \end{shadebox}
        \begin{proof}
            \quad\par
            $\bm{X}$と$\bm{Y}$の確率密度関数が一致することを示せば良い。
            両者の確率密度関数を各々$p_{\bm{X}},p_{\bm{Y}}$とする。
            任意の点$\bm{y}_0\in\realNumbers^n$における$p_{\bm{Y}}$を計算するために、$\Omega$を、$\bm{y}_0$を含む任意の領域とする。
            $\Omega$を直交変換$P$の逆写像で写した領域を$P^\top(\Omega)$と表記すると
            \begin{align}
                p_{\bm{Y}}(\bm{y}_0) &\coloneqq \lim_{|\Omega|\to 0}\frac{\Pr{\bm{y}\in\Omega}}{|\Omega|} = \lim_{|\Omega|\to 0}\frac{\Pr{\bm{x}\in P^\top(\Omega)}}{|\Omega|} = \lim_{|\Omega|\to 0}\frac{1}{|\Omega|}\integrate{P^\top(\Omega)}{}{p_{\bm{X}}(\bm{x})}{}{\bm{x}} \nonumber\\
                &= \lim_{|\Omega|\to 0}\frac{1}{|\Omega|}\integrate{P^\top(\Omega)}{}{ \frac{1}{(2\pi\sigma^2)^{n/2}}\exp{\left(-\frac{\norm{\bm{x}}^2}{2\sigma^2}\right)} }{}{\bm{x}} \nonumber\\
                &= \lim_{|\Omega|\to 0}\frac{1}{|\Omega|}\integrate{P(P^\top(\Omega))}{}{ \frac{1}{(2\pi\sigma^2)^{n/2}}\exp{\left(-\frac{\norm{P^\top\bm{y}}^2}{2\sigma^2}\right)} |\det{P^\top}|}{}{\bm{y}} \\
                &\quad \quad(\text{↑重積分の変数変換}\bm{x}=P^\top\bm{y}\textrm{を用いた}) \nonumber\\
                &= \lim_{|\Omega|\to 0}\frac{1}{|\Omega|}\integrate{\Omega}{}{ \frac{1}{(2\pi\sigma^2)^{n/2}}\exp{\left(-\frac{\norm{\bm{y}}^2}{2\sigma^2}\right)} }{}{\bm{y}} \quad(P\textrm{は直交行列だから}\det{P^\top}=\pm 1,\;\norm{P^\top\bm{y}}=\norm{\bm{y}}) \nonumber\\
                &= \lim_{|\Omega|\to 0}\frac{1}{|\Omega|}\integrate{\Omega}{}{ p_{\bm{X}}(\bm{y}) }{}{\bm{y}}
            \end{align}
            重積分の平均値の定理より、ある$\tilde{\bm{y}}\in\Omega$が存在して
            \[(1) = \lim_{|\Omega|\to 0}\frac{1}{|\Omega|}|\Omega|p_{\bm{X}}(\tilde{\bm{y}}) = \lim_{|\Omega|\to 0}p_{\bm{X}}(\tilde{\bm{y}}) = p_{\bm{X}}(\bm{y}_0)\]
        \end{proof}
    \section{多変量正規分布}
        \subsection{定義}
            $\Sigma$を正定値対称行列とし、対角化が$\diag{{\sigma_1}^2,\dotsc,{\sigma_n}^2}$であるとする。
            確率変数$X_1,\dotsc,X_n$の同時確率密度関数$f_{\bm{X}}(\bm{x})$が
            \[ f_{\bm{X}}(\bm{x}) = \frac{1}{(2\pi)^{n/2}|\Sigma|^{1/2}} \exp\left[ -\frac{1}{2}(\bm{x}-\bm{\mu})^\top\Sigma^{-1}(\bm{x}-\bm{\mu}) \right]\]
            であるとき、$X_1,\dotsc,X_n$が従う分布は\textbf{多変量正規分布}と呼ばれ、$N(\bm{\mu},\Sigma)$と表記される。
            期待値と共分散行列は$\E{}{\bm{X}}=\bm{\mu},\;\CovMat{\bm{X}} = \Sigma$である。
        \subsection{密度関数の全空間積分が1になることの確認}
            \label{subsec:多変量正規分布の密度関数の全空間積分が1になることの確認}
            $\Sigma$に関する仮定より、適当な直交行列$V$が存在して$\Sigma = V\Lambda V^\top,\;\Lambda=\diag{{\sigma_1}^2,\dotsc,{\sigma_n}^2}$と変形できる。
            そこで$\bm{x}-\bm{\mu} = V\bm{y}$すなわち$\bm{y} = V^\top(\bm{x}-\bm{\mu})$なる直交変換を行うと、求めたい積分は次のように変形できる。
            \begin{align*}
                \integrate{\realNumbers^n}{}{f_{\bm{X}}(\bm{x})}{}{\bm{x}} &= \integrate{\realNumbers^n}{}{\frac{1}{(2\pi)^{n/2}|\Sigma|^{1/2}}\exp\left(-\frac{1}{2}\bm{y}^\top V (V\Lambda V^\top)^{-1} V\bm{y}\right) |\det V|}{}{\bm{y}} \\
                &= \integrate{\realNumbers^n}{}{\frac{1}{\prod_{i=1}^n \sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2}\bm{y}^\top\Lambda^{-1}\bm{y}\right)}{}{\bm{y}} \quad (\because\;\text{は直交行列なので}|\det V|=1) \\
                &= \integrate{\realNumbers^n}{}{\prod_{i=1}^n\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}{y_i}^2\right)}{}{\bm{y}} = \prod_{i=1}^n \integrate{-\infty}{\infty}{\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}{y_i}^2\right)}{}{y_i} = 1
            \end{align*}
        \subsection{期待値の導出}
            \begin{proof}
                \quad\par
                先程の直交変換をここでも用いる。
                \begin{align*}
                    \E{}{\bm{X}} &= \integrate{\realNumbers^n}{}{\bm{x}f_{\bm{X}}(\bm{x})}{}{\bm{x}} = \integrate{\realNumbers^n}{}{(V\bm{y}+\bm{\mu})\frac{1}{\prod_{i=1}^n \sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2}\bm{y}^\top\Lambda^{-1}\bm{y}\right)}{}{\bm{y}} \\
                    &= \bm{\mu} + V\integrate{\realNumbers^n}{}{\bm{y}\frac{1}{\prod_{i=1}^n \sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2}\bm{y}^\top\Lambda^{-1}\bm{y}\right)}{}{\bm{y}} \\
                    &= \bm{\mu} + V\integrate{\realNumbers^n}{}{\bm{y}\prod_{i=1}^n\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}{y_i}^2\right)}{}{\bm{y}} \\
                    &= \bm{\mu} + V\integrate{y_n=-\infty}{\infty}{\dotsi\integrate{y_1=-\infty}{\infty}{\bm{y}\prod_{i=1}^n\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}{y_i}^2\right)}{}{y_1}\dotsi}{}{y_n}
                \end{align*}
                第二項目はベクトルの各インデックス毎に奇関数の対称積分が起こり、結局全体が0になることがわかる。
            \end{proof}
        \subsection{共分散行列の導出}
            \begin{proof}
                \quad\par
                \ref{subsec:多変量正規分布の密度関数の全空間積分が1になることの確認}で用いた直交変換をここでも用いる。
                \begin{align*}
                    \CovMat{\bm{X}} &= \E{}{(\bm{X}-\E{}{\bm{X}})(\bm{X}-\E{}{\bm{X}})^\top} = \integrate{\realNumbers^n}{}{(\bm{x}-\bm{\mu})(\bm{x}-\bm{\mu})^\top f_{\bm{X}}(\bm{x})}{}{\bm{x}} \\
                    &= \integrate{\realNumbers^n}{}{V\bm{y}\bm{y}^\top V^\top\prod_{i=1}^n\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}{y_i}^2\right)}{}{\bm{y}} \\
                    &= V\left(\integrate{\realNumbers^n}{}{\bm{y}\bm{y}^\top \prod_{i=1}^n\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}{y_i}^2\right)}{}{\bm{y}}\right)V^\top
                \end{align*}
                積分の部分に着目すると、非対角要素では奇関数の対称積分が起こり、0になる。
                第$k$対角成分は、$y_i(i\neq k)$に関する積分を先に実行することで
                \begin{align*}
                    \integrate{\realNumbers^n}{}{{y_k}^2\prod_{i=1}^n\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}{y_i}^2\right)}{}{\bm{y}} = \integrate{-\infty}{\infty}{{y_k}^2\frac{1}{\sqrt{2\pi}\sigma_k}\exp\left(-\frac{1}{2{\sigma_k}^2}{y_k}^2\right)}{}{y_k} = {\sigma_k}^2
                \end{align*}
                となることがわかる。
                よって
                \[ \CovMat{\bm{X}} = V\Lambda V^\top = \Sigma \]
            \end{proof}
        \subsection{特性関数は\texorpdfstring{$\phi_{\bm{X}}(\bm{t})=\exp(i\bm{t}^\top\bm{\mu} - \bm{t}^\top\Sigma\bm{t}/2)$}{φ\_X (t) = exp(i t\string^T μ - t\string^T Σ t/2)}}
            \begin{proof}
                \quad\par
                \ref{subsec:多変量正規分布の密度関数の全空間積分が1になることの確認}で用いた直交変換をここでも用いる。
                \begin{align*}
                    \phi_{\bm{X}}(\bm{t}) &= \E{}{\exp(i\bm{t}^\top \bm{X})} = \integrate{\realNumbers^n}{}{\exp(i\bm{t}^\top\bm{x})f_{\bm{X}}(\bm{x})}{}{\bm{x}} \\
                    &= \integrate{\realNumbers^n}{}{\frac{1}{(2\pi)^{n/2}|\Sigma|^{1/2}}\exp(i\bm{t}^\top\bm{x})\exp\left(-\frac{1}{2}(\bm{x}-\bm{\mu})^\top\Sigma^{-1}(\bm{x}-\bm{\mu})\right)}{}{\bm{x}} \\
                    &= \integrate{\realNumbers^n}{}{\frac{1}{(2\pi)^{n/2}\prod_{i=1}^n\sigma_i}\exp(i\bm{t}^\top(V\bm{y}+\bm{\mu}))\exp\left(-\frac{1}{2}\bm{y}^\top\Lambda^{-1}\bm{y}\right)|\det V|}{}{\bm{y}} \\
                    &= \exp(i\bm{t}^\top\bm{\mu})\integrate{\realNumbers^n}{}{\frac{1}{(2\pi)^{n/2}\prod_{i=1}^n\sigma_i}\exp\left(-\frac{1}{2}\bm{y}^\top\Lambda^{-1}\bm{y} + i\bm{t}^\top V\bm{y}\right)}{}{\bm{y}} \\
                    &= \exp(i\bm{t}^\top\bm{\mu})\integrate{\realNumbers^n}{}{\frac{1}{(2\pi)^{n/2}\prod_{i=1}^n\sigma_i}\exp\left(-\frac{1}{2}\sum_{i=1}^n\frac{1}{{\sigma_i}^2}{y_i}^2 + i\bm{t}^\top V\bm{y}\right)}{}{\bm{y}}
                \end{align*}
                ここで$c_i \coloneqq (\bm{t}^\top V)[i]$とするとexpの引数は次のように変形できる。
                \begin{align*}
                    & -\frac{1}{2}\sum_{i=1}^n\frac{1}{{\sigma_i}^2}{y_i}^2 + i\bm{t}^\top V\bm{y} = -\frac{1}{2}\sum_{i=1}^n\frac{1}{{\sigma_i}^2}{y_i}^2 + i\sum_{i=1}^n c_iy_i = -\frac{1}{2}\sum_{i=1}^n\frac{1}{{\sigma_i}^2}\left({y_i}^2 - i2{\sigma_i}^2c_iy_i\right) \\
                    =& -\frac{1}{2}\sum_{i=1}^n\frac{1}{{\sigma_i}^2}\left[\left(y_i - i{\sigma_i}^2c_i\right)^2 + {\sigma_i}^4{c_i}^2\right] = -\frac{1}{2}\sum_{i=1}^n{\sigma_i}^2{c_i}^2 - \frac{1}{2}\sum_{i=1}^n\frac{1}{{\sigma_i}^2}\left(y_i - i{\sigma_i}^2c_i\right)^2
                \end{align*}
                さらに次式が成り立つ。
                \[ \sum_{i=1}^n{\sigma_i}^2{c_i}^2 = [c_1,\dotsc,c_n]\Lambda[c_1,\dotsc,c_n]^\top = \bm{t}^\top V\Lambda V^\top\bm{t} = \bm{t}^\top\Sigma\bm{t} \]
                よって
                \begin{align*}
                    \phi_{\bm{X}}(t) &= \exp\left(i\bm{t}^\top\bm{\mu} - \frac{1}{2}\bm{t}^\top\Sigma\bm{t}\right)\integrate{\realNumbers^n}{}{\frac{1}{(2\pi)^{n/2}\prod_{i=1}^n\sigma_i}\exp\left(-\frac{1}{2}\sum_{i=1}^n\frac{1}{{\sigma_i}^2}\left(y_i - i{\sigma_i}^2c_i\right)^2\right)}{}{\bm{y}} \\
                    &= \exp\left(i\bm{t}^\top\bm{\mu} - \frac{1}{2}\bm{t}^\top\Sigma\bm{t}\right)\integrate{\realNumbers^n}{}{\prod_{i=1}^n\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}\left(y_i - i{\sigma_i}^2c_i\right)^2\right)}{}{\bm{y}} \\
                    &= \exp\left(i\bm{t}^\top\bm{\mu} - \frac{1}{2}\bm{t}^\top\Sigma\bm{t}\right)\prod_{i=1}^n I_i\quad \left(I_i \coloneqq \integrate{-\infty}{\infty}{\frac{1}{\sqrt{2\pi}\sigma_i}\exp\left(-\frac{1}{2{\sigma_i}^2}\left(y_i - i{\sigma_i}^2c_i\right)^2\right)}{}{y_i}\right)
                \end{align*}
                $I_i$は複素積分により1である(矩形閉路ABCDA: $A=(-L,0),\;B=(L,0),\;C=(L,i{\sigma_i}^2c_i),\;D=(-L,i{\sigma_i}^2c_i)$を用いて$L\to\infty$とするとよい)ので
                \[ \phi_{\bm{X}}(t) = \exp\left(i\bm{t}^\top\bm{\mu} - \frac{1}{2}\bm{t}^\top\Sigma\bm{t}\right) \]
            \end{proof}
        \subsection{特性関数を用いた期待値と分散の導出}
            \begin{proof}
                \quad\par
                まず、期待値については$\nabla_{\bm{t}}\phi_{\bm{X}}(\bm{t}) = (i\bm{\mu}-\Sigma\bm{t})\phi_{\bm{X}}(\bm{t})$より$\E{}{\bm{X}} = -i\nabla_{\bm{t}}\phi_{\bm{X}}(\bm{0}) = \bm{\mu}$である。
                次に、共分散行列については、
                \[ \nabla_{\bm{t}}^2\phi_{\bm{X}}(\bm{t}) = (-\Sigma + (i\bm{\mu}-\Sigma\bm{t})(i\bm{\mu}-\Sigma\bm{t})^\top)\phi_{\bm{X}}(\bm{t}) \]
                より$\E{}{\bm{X}\bm{X}^\top} = -\nabla_{\bm{t}}^2\phi_{\bm{X}}(\bm{0}) = \Sigma + \bm{\mu}\bm{\mu}^\top$であるので
                $\CovMat{\bm{X}} = \E{}{\bm{X}\bm{X}^\top} - \E{}{\bm{X}}\E{}{\bm{X}}^\top = \Sigma$
            \end{proof}
        \subsection{線形写像 : \texorpdfstring{$A\bm{X}+\bm{b} \sim N(A\bm{\mu}+\bm{b}, A\Sigma A^\top)$}{AX+b ～ N(Aμ+b, AΣA\string^T)}}
            \begin{proof}
                \quad\par
                特性関数を用いて証明する。
                \begin{align*}
                    \varphi_{A\bm{X}+\bm{b}}(\bm{t}) &= \E{}{\exp(i\bm{t}^\top(A\bm{X}+\bm{b}))} = \exp(i\bm{t}^\top\bm{b})\E{}{\exp(i\bm{t}^\top A\bm{X})} = \exp(i\bm{t}^\top\bm{b})\E{}{\exp(i(A^\top\bm{t})^\top\bm{X})} \\
                    &= \exp(i\bm{t}^\top\bm{b})\varphi_{\bm{X}}(A^\top\bm{t}) = \exp(i\bm{t}^\top\bm{b})\exp\left(i(A^\top\bm{t})^\top\bm{\mu} - \frac{1}{2}(A^\top\bm{t})^\top\Sigma(A^\top\bm{t})\right) \\
                    &= \exp(i\bm{t}^\top\bm{b})\exp\left(i\bm{t}^\top(A\bm{\mu}) - \frac{1}{2}\bm{t}^\top(A\Sigma A^\top)\bm{t}\right) = \exp\left(i\bm{t}^\top(A\bm{\mu}+\bm{b}) - \frac{1}{2}\bm{t}^\top(A\Sigma A^\top)\bm{t}\right)
                \end{align*}
                これは期待値$A\bm{\mu}+\bm{b}$,\;共分散行列$A\Sigma A^\top$の多変量正規分布の特性関数であるから、$A\bm{X}+\bm{b}$がこの多変量正規分布に従うことがわかる。
            \end{proof}
