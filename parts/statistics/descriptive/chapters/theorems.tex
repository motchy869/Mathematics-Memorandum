\chapter{諸定理}
    \section{分散共分散行列は半正定。特に分散が正なら正定。}
        \label{分散共分散行列の定値性}
        \begin{shadebox}
            $D$個の$n$次元ベクトルから成るデータ集合$\DS \coloneqq \{\bm{x}_d\in\realNumbers^n\}_{d=1}^D$の分散共分散行列$\Sigma_{\DS}$は半正定である。
            特に$\Var{}{\DS}>0$ならば$\Sigma_{\DS}$は正定である。
        \end{shadebox}
        \begin{proof}
            \quad\par
            任意の$\bm{v}\in\realNumbers^n\setminus{\bm{0}_n}$に対して
            \[\bm{v}^\top\Sigma_{\DS}\bm{v} = \frac{1}{D}\sum_{d=1}^D{\bm{v}^\top(\bm{x}_d-\overline{\bm{x}})(\bm{x}_d-\overline{\bm{x}})^\top\bm{v}} = \frac{1}{D}\sum_{d=1}^D{\norm{\bm{v}^\top(\bm{x}_d-\overline{\bm{x}})}^2} \geq 0\]
            であるから少なくとも半正定である。さらに
            \begin{align*}
                \textrm{等号成立} &\iff \norm{\bm{v}^\top(\bm{x}_d-\overline{\bm{x}})}^2 = 0,\;^\forall d\in\{1,\dotsc,D\},\;^\forall \bm{v}\in\realNumbers^n\setminus{\bm{0}_n} \iff \bm{x}_d = \overline{\bm{x}},\;^\forall d\in{1,\dotsc,D} \\
                &\iff \Var{}{\DS} = 0
            \end{align*}
        \end{proof}
